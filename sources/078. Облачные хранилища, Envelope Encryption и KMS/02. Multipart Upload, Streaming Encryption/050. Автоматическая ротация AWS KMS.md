# Автоматическая ротация AWS KMS: Версионирование ключей и обратная совместимость

## Условие

Настройте автоматическую ротацию ключа в AWS KMS (или эмуляцию через версионирование). Реализуйте механизм, при котором старые DEK расшифровываются старой версией ключа, но новые шифруются новой версией.

### Исходная формулировка

**Ротация ключей KMS.**: Настройте автоматическую ротацию ключа в AWS KMS (или эмуляцию через версионирование). Реализуйте механизм, при котором старые DEK расшифровываются старой версией ключа, но новые шифруются новой версией.

## Теория

При включении автоматической ротации в AWS KMS (`EnableKeyRotation`):
1. KMS ежегодно генерирует новую криптографическую версию мастер-ключа внутри HSM.
2. ARN и идентификатор ключа `KeyId` **не меняются**.
3. **Обратная совместимость (Backwards Compatibility):**
   - Все новые вызовы `GenerateDataKey` или `Encrypt` используют **актуальную последнюю версию** ключа.
   - Все исторические данные, зашифрованные старыми версиями ключа, автоматически расшифровываются соответствующей старой версией (KMS хранит все предыдущие поколения внутри HSM).
   - Приложению не требуется выполнять перешифровку исторических данных!

## Решение по шагам

1. Спроектируйте модель версионированного ключа `VersionedKMSKey` с массивом поколений `versions`.
2. Реализуйте метод `RotateKey()`, добавляющий новую версию и делающий ее активной для записи.
3. Реализуйте метод `Encrypt` (использует последнюю версию) и `Decrypt` (использует версию из заголовка).
4. Продемонстрируйте корректную расшифровку старых данных после многократной ротации.

## Код решения

**`versioned_kms_rotation.go`**

```go
package main

import (
	"crypto/aes"
	"crypto/cipher"
	"crypto/rand"
	"encoding/binary"
	"errors"
	"fmt"
	"io"
	"log"
	"sync"
)

// VersionedMasterKey эмулирует автоматическую ротацию поколений в AWS KMS
type VersionedMasterKey struct {
	mu            sync.RWMutex
	keyID         string
	versions      map[uint32][]byte // versionNumber -> raw 32-byte key
	activeVersion uint32
}

func NewVersionedMasterKey(keyID string) *VersionedMasterKey {
	initialKey := make([]byte, 32)
	io.ReadFull(rand.Reader, initialKey)

	return &VersionedMasterKey{
		keyID:         keyID,
		versions:      map[uint32][]byte{1: initialKey},
		activeVersion: 1,
	}
}

// RotateKey генерирует новую версию мастер-ключа
func (k *VersionedMasterKey) RotateKey() uint32 {
	k.mu.Lock()
	defer k.mu.Unlock()

	newKey := make([]byte, 32)
	io.ReadFull(rand.Reader, newKey)

	k.activeVersion++
	k.versions[k.activeVersion] = newKey
	return k.activeVersion
}

// Encrypt всегда использует последнюю активную версию ключа
func (k *VersionedMasterKey) Encrypt(plaintext []byte) ([]byte, error) {
	k.mu.RLock()
	v := k.activeVersion
	rawKey := k.versions[v]
	k.mu.RUnlock()

	block, _ := aes.NewCipher(rawKey)
	gcm, _ := cipher.NewGCM(block)
	nonce := make([]byte, gcm.NonceSize())
	io.ReadFull(rand.Reader, nonce)

	ciphertext := gcm.Seal(nil, nonce, plaintext, nil)

	// Формат блоба: [Version (uint32)][Nonce (12B)][Ciphertext]
	result := make([]byte, 4+len(nonce)+len(ciphertext))
	binary.BigEndian.PutUint32(result[:4], v)
	copy(result[4:4+len(nonce)], nonce)
	copy(result[4+len(nonce):], ciphertext)

	return result, nil
}

// Decrypt автоматически выбирает правильную историческую версию ключа из заголовка блоба
func (k *VersionedMasterKey) Decrypt(blob []byte) ([]byte, uint32, error) {
	if len(blob) < 16 {
		return nil, 0, errors.New("blob too short")
	}

	v := binary.BigEndian.Uint32(blob[:4])

	k.mu.RLock()
	rawKey, exists := k.versions[v]
	k.mu.RUnlock()

	if !exists {
		return nil, 0, fmt.Errorf("key version %d not found (retired or deleted)", v)
	}

	block, _ := aes.NewCipher(rawKey)
	gcm, _ := cipher.NewGCM(block)
	nSize := gcm.NonceSize()

	nonce := blob[4 : 4+nSize]
	ciphertext := blob[4+nSize:]

	plaintext, err := gcm.Open(nil, nonce, ciphertext, nil)
	if err != nil {
		return nil, 0, fmt.Errorf("decrypt failed: %w", err)
	}

	return plaintext, v, nil
}

func main() {
	kmsKey := NewVersionedMasterKey("arn:aws:kms:eu-central-1:123456:key/billing-master")
	fmt.Printf("1. Инициализирован ключ. Текущая активная версия: v%d\n", kmsKey.activeVersion)

	// Шифруем запись версией 1
	doc2024 := []byte("Financial Invoice 2024 - Total: $12,000")
	blob2024, err := kmsKey.Encrypt(doc2024)
	if err != nil {
		log.Fatalf("Encrypt error: %v", err)
	}
	fmt.Printf("2. Запись 2024 года зашифрована активной версией v%d\n", kmsKey.activeVersion)

	// Наступает плановая ежегодная ротация ключа KMS
	newV := kmsKey.RotateKey()
	fmt.Printf("\n--- Произошла автоматическая ротация KMS: новая активная версия v%d ---\n\n", newV)

	// Шифруем новую запись версией 2
	doc2025 := []byte("Financial Invoice 2025 - Total: $18,500")
	blob2025, err := kmsKey.Encrypt(doc2025)
	if err != nil {
		log.Fatalf("Encrypt error: %v", err)
	}
	fmt.Printf("3. Запись 2025 года зашифрована новой активной версией v%d\n", kmsKey.activeVersion)

	// Проверяем прозрачную расшифровку обоих документов
	res2024, usedV1, _ := kmsKey.Decrypt(blob2024)
	res2025, usedV2, _ := kmsKey.Decrypt(blob2025)

	fmt.Printf("\n4. Расшифровка записи 2024: '%s' (автоматически использована историческая версия v%d)\n", string(res2024), usedV1)
	fmt.Printf("5. Расшифровка записи 2025: '%s' (автоматически использована актуальная версия v%d)\n", string(res2025), usedV2)

	if string(res2024) == string(doc2024) && string(res2025) == string(doc2025) {
		fmt.Println("Успех: Автоматическая ротация KMS сохраняет 100% обратную совместимость!")
	}
}
```

## Под капотом

Механика реализации 'Автоматическая ротация AWS KMS: Версионирование ключей и обратная совместимость':
В рантайме Go и сетевом стеке низкоуровневые операции ввода-вывода абстрагируются через интерфейсы `io.Reader` и `io.Writer`. При работе с криптографическими алгоритмами задействуются аппаратные инструкции процессора (AES-NI / CLMUL на x86-64 и ARM Cryptography Extensions), что обеспечивает скорость обработки до 3–5 ГБ/с на ядро CPU с околонулевым оверхедом планировщика GMP.

## Подводные камни

Типичная ошибка при проектировании: утечка неуправляемых ресурсов и памяти. При работе с S3 сетевыми потоками обязателен вызов `defer response.Body.Close()`, иначе сокеты TCP останутся в состоянии CLOSE_WAIT, что приведет к исчерпанию лимита файловых дескрипторов ОС (`too many open files`). Также недопустимо сохранять открытые криптографические ключи в глобальных структурах данных.

## Лучшие практики

- Включайте `EnableKeyRotation` в AWS KMS для всех Customer Managed Keys для соответствия стандарту PCI DSS
- Помните: ротация KMS не перешифровывает старые данные, она лишь генерирует новый ключ для последующих записей
- Не удаляйте предыдущие версии ключей, пока в хранилище остаются зашифрованные ими объекты
