# Маршрутизация комнат по URL: динамический путь /chat/{roomID} и автоматическая изоляция

## Условие

Реализуйте **Room/Channel паттерн**: клиенты могут "заходить" в комнаты (`/chat/room-1`) и получать сообщения только своей комнаты.

## Теория

Привязка комнаты к URL пути запроса:
- Вместо отправки JSON сообщений `"join"` комната может задаваться прямо в URL WebSocket подключения:
  - `ws://example.com/chat/room-1`
  - `ws://example.com/chat/room-2`
- Преимущества:
  - Простая интеграция с веб-страницами (URL вкладки браузера определяет адрес сокета).
  - Балансировщик нагрузки (Nginx/HAProxy) может балансировать клиентов одной комнаты на конкретные серверные шарды по URL пути.

## Решение по шагам

1. Создайте HTTP обработчик с разбором URL пути `/chat/{roomID}`.
2. Привяжите соединение к указанной комнате.
3. Проверьте изоляцию сообщений между `/chat/room-1` и `/chat/room-2`.
4. Протестируйте корректный дисконнект.

## Код решения

**`url_room_handler_test.go`**

```go
package main

import (
	"fmt"
	"net/http"
	"net/http/httptest"
	"strings"
	"sync"
	"testing"

	"github.com/gorilla/websocket"
)

type URLRoomServer struct {
	mu    sync.Mutex
	rooms map[string]map[*websocket.Conn]bool
}

func NewURLRoomServer() *URLRoomServer {
	return &URLRoomServer{rooms: make(map[string]map[*websocket.Conn]bool)}
}

func (s *URLRoomServer) ServeHTTP(w http.ResponseWriter, r *http.Request) {
	// Извлекаем имя комнаты из пути /chat/{roomID}
	parts := strings.Split(strings.Trim(r.URL.Path, "/"), "/")
	if len(parts) != 2 || parts[0] != "chat" {
		http.Error(w, "Ожидается путь /chat/{roomID}", http.StatusBadRequest)
		return
	}
	roomID := parts[1]

	upgrader := websocket.Upgrader{}
	conn, err := upgrader.Upgrade(w, r, nil)
	if err != nil {
		return
	}
	defer conn.Close()

	s.mu.Lock()
	if _, ok := s.rooms[roomID]; !ok {
		s.rooms[roomID] = make(map[*websocket.Conn]bool)
	}
	s.rooms[roomID][conn] = true
	s.mu.Unlock()

	defer func() {
		s.mu.Lock()
		delete(s.rooms[roomID], conn)
		s.mu.Unlock()
	}()

	for {
		mt, msg, err := conn.ReadMessage()
		if err != nil {
			break
		}
		// Рассылаем строго внутри этой комнаты
		s.mu.Lock()
		for member := range s.rooms[roomID] {
			_ = member.WriteMessage(mt, msg)
		}
		s.mu.Unlock()
	}
}

func TestURLRoomRouting(t *testing.T) {
	srv := NewURLRoomServer()
	ts := httptest.NewServer(srv)
	defer ts.Close()

	wsURL := "ws" + strings.TrimPrefix(ts.URL, "http")

	// Клиент 1 подключается к /chat/room-alpha
	cAlpha, _, _ := websocket.DefaultDialer.Dial(wsURL+"/chat/room-alpha", nil)
	defer cAlpha.Close()

	// Клиент 2 подключается к /chat/room-beta
	cBeta, _, _ := websocket.DefaultDialer.Dial(wsURL+"/chat/room-beta", nil)
	defer cBeta.Close()

	// Отправляем сообщение в room-alpha
	_ = cAlpha.WriteMessage(websocket.TextMessage, []byte("Сообщение для Альфы"))

	_, msgAlpha, _ := cAlpha.ReadMessage()
	if string(msgAlpha) != "Сообщение для Альфы" {
		t.Fatalf("Alpha не получила свое сообщение: %s", string(msgAlpha))
	}

	fmt.Println("Маршрутизация комнат по URL пути /chat/{roomID} успешно протестирована!")
}
```

## Под капотом

В Go 1.22+ стандартный роутер `http.ServeMux` поддерживает сопоставление путей `mux.HandleFunc("GET /chat/{roomID}", handler)`, делая извлечение переменных пути стандартным вызовом `r.PathValue("roomID")`.

## Подводные камни

Не санитизировать имя комнаты: клиент может передать `roomID = "../admin"`, что при использовании в путях файловой системы приведет к уязвимости Directory Traversal.

## На собеседовании

**Вопрос с собеседования:** «Как реализовать sticky routing (привязку) комнат чата в Nginx?»
**Ответ:** Использовать директиву `hash $room_id consistent` в блоке `upstream`. Имя комнаты извлекается из регулярного выражения URL (`rewrite ^/chat/(?<room_id>[^/]+)$`), и все запросы к одной комнате попадают строго на один и тот же бэкенд, исключая межнодовые сетевые вызовы.
