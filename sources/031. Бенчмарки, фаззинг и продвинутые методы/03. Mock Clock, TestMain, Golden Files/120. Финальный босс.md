## Теория

Комплексная архитектура микросервиса URL Shortener корпоративного уровня:
1. **Кодирование Base62 (Zero-Alloc):** алгоритм преобразования 64-битного ID в компактный алфавитно-числовой хэш `[0-9a-zA-Z]`.
2. **Fuzzing валидатора URL и Base62:** защита от вредоносных символов, `javascript:` схем и SSRF-атак.
3. **HTTP API Middleware:** аутентификация Bearer Token и Rate Limiting на уровне промежуточного слоя.
4. **Конкурентная защита:** использование `sync.Map` или мьютексов для исключения дублирования слагов под флагом `-race`.
5. **Тестирование и Бенчмарки:** покрытие 90%+, замер скорости генерации кодов (<20 нс).

## Решение по шагам

1. Реализуйте алгоритм Base62 Encode/Decode.
2. Создайте потокобезопасное хранилище URL.

## Код решения

**`shortener_core.go`**

```go
package main

import (
	"errors"
	"net/url"
	"strings"
	"sync"
)

const base62Alphabet = "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ"

// Base62Encode кодирует число в компактную строку
func Base62Encode(n uint64) string {
	if n == 0 {
		return "0"
	}
	var b [11]byte
	i := len(b)
	for n > 0 {
		i--
		b[i] = base62Alphabet[n%62]
		n /= 62
	}
	return string(b[i:])
}

// Base62Decode декодирует строку обратно в uint64
func Base62Decode(s string) (uint64, error) {
	if s == "" {
		return 0, errors.New("пустая строка")
	}
	var res uint64
	for i := 0; i < len(s); i++ {
		c := s[i]
		idx := strings.IndexByte(base62Alphabet, c)
		if idx == -1 {
			return 0, errors.New("недопустимый символ в base62")
		}
		res = res*62 + uint64(idx)
	}
	return res, nil
}

// ValidateTargetURL проверяет безопасность адреса
func ValidateTargetURL(raw string) error {
	u, err := url.ParseRequestURI(raw)
	if err != nil {
		return errors.New("некорректный URL")
	}
	if u.Scheme != "http" && u.Scheme != "https" {
		return errors.New("разрешены только http/https схемы")
	}
	if u.Host == "" {
		return errors.New("отсутствует хост")
	}
	return nil
}

type SafeURLStore struct {
	mu   sync.RWMutex
	urls map[string]string
}

func NewSafeURLStore() *SafeURLStore {
	return &SafeURLStore{urls: make(map[string]string)}
}

func (s *SafeURLStore) Put(slug, target string) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	if _, exists := s.urls[slug]; exists {
		return errors.New("slug_conflict")
	}
	s.urls[slug] = target
	return nil
}

func (s *SafeURLStore) Get(slug string) (string, bool) {
	s.mu.RLock()
	defer s.mu.RUnlock()
	val, ok := s.urls[slug]
	return val, ok
}
```

**`shortener_suite_test.go`**

```go
package main

import (
	"fmt"
	"net/http"
	"net/http/httptest"
	"sync"
	"testing"
)

// 1. Бенчмарк кодирования Base62
var sinkSlug string

func BenchmarkBase62Encode(b *testing.B) {
	b.ReportAllocs()
	var s string
	for i := 0; i < b.N; i++ {
		s = Base62Encode(1005002026)
	}
	sinkSlug = s
}

// 2. Фаззинг Base62 кодека
func FuzzBase62Codec(f *testing.F) {
	f.Add(uint64(0))
	f.Add(uint64(100))
	f.Add(uint64(999999999))

	f.Fuzz(func(t *testing.T, val uint64) {
		encoded := Base62Encode(val)
		decoded, err := Base62Decode(encoded)
		if err != nil {
			t.Fatalf("Ошибка декодирования: %v", err)
		}
		if decoded != val {
			t.Fatalf("Инвариант Base62 нарушен! Orig: %d, Decoded: %d", val, decoded)
		}
	})
}

// 3. Стресс-тест конкурентности и защита от гонок данных
func TestConcurrentSlugConflictHandling(t *testing.T) {
	store := NewSafeURLStore()
	var wg sync.WaitGroup

	const numGoroutines = 50
	const targetSlug = "shared_link"

	successCount := 0
	var mu sync.Mutex

	wg.Add(numGoroutines)
	for i := 0; i < numGoroutines; i++ {
		go func(id int) {
			defer wg.Done()
			err := store.Put(targetSlug, fmt.Sprintf("https://example.com/user/%d", id))
			if err == nil {
				mu.Lock()
				successCount++
				mu.Unlock()
			}
		}(i)
	}

	wg.Wait()

	// Ровно одна горутина обязана выиграть регистрацию слага!
	if successCount != 1 {
		t.Fatalf("Инвариант нарушен: %d горутин успешно перезаписали один slug!", successCount)
	}

	// Проверка чтения
	url, ok := store.Get(targetSlug)
	if !ok || url == "" {
		t.Fatal("Не удалось прочитать созданный URL")
	}
}

// 4. HTTP API интеграционный тест с httptest
func TestHTTPShortenerFlow(t *testing.T) {
	store := NewSafeURLStore()
	_ = store.Put("doc", "https://golang.org/doc")

	handler := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		slug := r.URL.Path[1:] // Отрезаем ведущий слэш
		target, found := store.Get(slug)
		if !found {
			http.NotFound(w, r)
			return
		}
		http.Redirect(w, r, target, http.StatusMovedPermanently)
	})

	req := httptest.NewRequest("GET", "/doc", nil)
	rec := httptest.NewRecorder()

	handler.ServeHTTP(rec, req)

	if rec.Code != http.StatusMovedPermanently {
		t.Fatalf("status = %d; want 301 Redirect", rec.Code)
	}

	loc := rec.Header().Get("Location")
	if loc != "https://golang.org/doc" {
		t.Fatalf("Location = %q; want https://golang.org/doc", loc)
	}
}
```
