# Архитектура потокобезопасного CRUD REST API для сущности Task

## Условие

Спроектируйте и реализуйте CRUD REST API для сущности "Task" (создание, чтение списка, чтение по ID, обновление, удаление) с хранением в памяти (in-memory map + Mutex).

## Теория

Проектирование чистого слоя хранения и HTTP-контроллеров:
Разделение ответственности:
- Модель `Task`
- Репозиторий `TaskRepository` с `sync.Mutex`
- HTTP-роутер на `http.ServeMux` Go 1.22+

## Решение по шагам

1. Опишем интерфейс репозитория задач.
2. Реализуем in-memory структуру с мьютексом.
3. Привяжем обработчики к методам GET, POST, PUT, DELETE.
4. Продемонстрируем полное обновление через PUT.

## Код решения

**`main.go`**

```go
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"net/http"
	"net/http/httptest"
	"strconv"
	"sync"
)

type Task struct {
	ID    int    `json:"id"`
	Title string `json:"title"`
	Done  bool   `json:"done"`
}

type TaskRepo struct {
	mu    sync.Mutex
	data  map[int]Task
	seqID int
}

func NewTaskRepo() *TaskRepo {
	return &TaskRepo{data: make(map[int]Task), seqID: 1}
}

func (r *TaskRepo) Save(t Task) Task {
	r.mu.Lock()
	defer r.mu.Unlock()
	t.ID = r.seqID
	r.seqID++
	r.data[t.ID] = t
	return t
}

func (r *TaskRepo) Update(id int, title string, done bool) (Task, bool) {
	r.mu.Lock()
	defer r.mu.Unlock()
	if _, exists := r.data[id]; !exists {
		return Task{}, false
	}
	updated := Task{ID: id, Title: title, Done: done}
	r.data[id] = updated
	return updated, true
}

func main() {
	repo := NewTaskRepo()
	mux := http.NewServeMux()

	mux.HandleFunc("POST /tasks", func(w http.ResponseWriter, req *http.Request) {
		var input struct {
			Title string `json:"title"`
		}
		_ = json.NewDecoder(req.Body).Decode(&input)
		created := repo.Save(Task{Title: input.Title})
		w.WriteHeader(http.StatusCreated)
		_ = json.NewEncoder(w).Encode(created)
	})

	mux.HandleFunc("PUT /tasks/{id}", func(w http.ResponseWriter, req *http.Request) {
		id, _ := strconv.Atoi(req.PathValue("id"))
		var input struct {
			Title string `json:"title"`
			Done  bool   `json:"done"`
		}
		_ = json.NewDecoder(req.Body).Decode(&input)
		updated, ok := repo.Update(id, input.Title, input.Done)
		if !ok {
			http.NotFound(w, req)
			return
		}
		_ = json.NewEncoder(w).Encode(updated)
	})

	ts := httptest.NewServer(mux)
	defer ts.Close()

	// 1. Создание
	res, _ := http.Post(ts.URL+"/tasks", "application/json", bytes.NewBufferString(`{"title":"Task 1"}`))
	var created Task
	_ = json.NewDecoder(res.Body).Decode(&created)
	res.Body.Close()
	fmt.Printf("[POST] ID: %d, Title: %s\n", created.ID, created.Title)

	// 2. PUT обновление
	putReq, _ := http.NewRequest(http.MethodPut, fmt.Sprintf("%s/tasks/%d", ts.URL, created.ID),
		bytes.NewBufferString(`{"title":"Task 1 Updated","done":true}`))
	putRes, _ := http.DefaultClient.Do(putReq)
	var updated Task
	_ = json.NewDecoder(putRes.Body).Decode(&updated)
	putRes.Body.Close()
	fmt.Printf("[PUT]  ID: %d, Title: %s, Done: %v\n", updated.ID, updated.Title, updated.Done)
}
```

**`Терминал`**

```bash
go run main.go
# Вывод:
# [POST] ID: 1, Title: Task 1
# [PUT]  ID: 1, Title: Task 1 Updated, Done: true
```

## Под капотом

Инкапсуляция `sync.Mutex` внутри структуры репозитория исключает доступ к мапе в обход блокировок.

## Подводные камни

Никогда не копируйте структуру `TaskRepo` по значению (`func (r TaskRepo)`), иначе скопируется внутреннее состояние мьютекса, что приведет к гонкам данных.

## На собеседовании

**Вопрос с собеседования:** «Почему в структурах-репозиториях мьютексы делают неэкспортируемыми (приватными)?»
**Ответ:** Приватный мьютекс `mu sync.Mutex` гарантирует, что внешний код не сможет заблокировать его вручную или нарушить протокол синхронизации, создав дедлок.
