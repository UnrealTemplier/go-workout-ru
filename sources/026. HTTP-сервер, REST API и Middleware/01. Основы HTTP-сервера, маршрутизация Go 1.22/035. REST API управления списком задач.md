# REST API управления списком задач (Task Tracker CRUD) на ServeMux

## Условие

Создайте REST API для управления списком задач (in-memory): `GET /tasks`, `POST /tasks`, `GET /tasks/{id}`, `DELETE /tasks/{id}`. Используйте стандартный `http.ServeMux` с поддержкой методов.

## Теория

Проектирование чистых REST API коллекций:
- `GET /tasks` — получить список задач.
- `POST /tasks` — создать новую задачу.
- `GET /tasks/{id}` — получить задачу по ID.
- `DELETE /tasks/{id}` — удалить задачу.

## Решение по шагам

1. Опишем структуру `Task { ID, Title, Completed }`.
2. Реализуем потокобезопасный `TaskStore`.
3. Зарегистрируем маршруты в `http.NewServeMux()`.
4. Протестируем создание, чтение и удаление задачи.

## Код решения

**`main.go`**

```go
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"net/http/httptest"
	"strconv"
	"sync"
)

type Task struct {
	ID        int    `json:"id"`
	Title     string `json:"title"`
	Completed bool   `json:"completed"`
}

type TaskManager struct {
	mu     sync.RWMutex
	tasks  map[int]Task
	nextID int
}

func NewTaskManager() *TaskManager {
	return &TaskManager{tasks: make(map[int]Task), nextID: 1}
}

func (m *TaskManager) Add(title string) Task {
	m.mu.Lock()
	defer m.mu.Unlock()
	t := Task{ID: m.nextID, Title: title, Completed: false}
	m.tasks[t.ID] = t
	m.nextID++
	return t
}

func (m *TaskManager) List() []Task {
	m.mu.RLock()
	defer m.mu.RUnlock()
	res := make([]Task, 0, len(m.tasks))
	for _, t := range m.tasks {
		res = append(res, t)
	}
	return res
}

func (m *TaskManager) Get(id int) (Task, bool) {
	m.mu.RLock()
	defer m.mu.RUnlock()
	t, ok := m.tasks[id]
	return t, ok
}

func (m *TaskManager) Delete(id int) bool {
	m.mu.Lock()
	defer m.mu.Unlock()
	if _, ok := m.tasks[id]; !ok {
		return false
	}
	delete(m.tasks, id)
	return true
}

func main() {
	tm := NewTaskManager()
	mux := http.NewServeMux()

	mux.HandleFunc("GET /tasks", func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "application/json")
		_ = json.NewEncoder(w).Encode(tm.List())
	})

	mux.HandleFunc("POST /tasks", func(w http.ResponseWriter, r *http.Request) {
		var input struct {
			Title string `json:"title"`
		}
		if err := json.NewDecoder(r.Body).Decode(&input); err != nil || input.Title == "" {
			http.Error(w, "некорректный заголовок задачи", http.StatusBadRequest)
			return
		}
		created := tm.Add(input.Title)
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusCreated)
		_ = json.NewEncoder(w).Encode(created)
	})

	mux.HandleFunc("GET /tasks/{id}", func(w http.ResponseWriter, r *http.Request) {
		id, _ := strconv.Atoi(r.PathValue("id"))
		t, ok := tm.Get(id)
		if !ok {
			http.NotFound(w, r)
			return
		}
		w.Header().Set("Content-Type", "application/json")
		_ = json.NewEncoder(w).Encode(t)
	})

	mux.HandleFunc("DELETE /tasks/{id}", func(w http.ResponseWriter, r *http.Request) {
		id, _ := strconv.Atoi(r.PathValue("id"))
		if !tm.Delete(id) {
			http.NotFound(w, r)
			return
		}
		w.WriteHeader(http.StatusNoContent)
	})

	ts := httptest.NewServer(mux)
	defer ts.Close()

	// 1. Создание задачи
	res1, _ := http.Post(ts.URL+"/tasks", "application/json", bytes.NewBufferString(`{"title":"Изучить роутинг Go 1.22"}`))
	b1, _ := io.ReadAll(res1.Body)
	res1.Body.Close()
	fmt.Printf("[Создано]  %s", string(b1))

	// 2. Получение списка
	res2, _ := http.Get(ts.URL + "/tasks")
	b2, _ := io.ReadAll(res2.Body)
	res2.Body.Close()
	fmt.Printf("[Список]   %s", string(b2))

	// 3. Удаление
	req, _ := http.NewRequest(http.MethodDelete, ts.URL+"/tasks/1", nil)
	res3, _ := http.DefaultClient.Do(req)
	res3.Body.Close()
	fmt.Printf("[Удаление] Status: %d (No Content)\n", res3.StatusCode)
}
```

**`Терминал`**

```bash
go run main.go
# Вывод:
# [Создано]  {"id":1,"title":"Изучить роутинг Go 1.22","completed":false}
# [Список]   [{"id":1,"title":"Изучить роутинг Go 1.22","completed":false}]
# [Удаление] Status: 204 (No Content)
```

## Под капотом

Методы Go 1.22 маршрутизатора сопоставляются до парсинга параметров пути, обеспечивая максимальную скорость диспетчеризации.

## Подводные камни

Ответ 204 No Content по стандарту HTTP запрещает отправку тела ответа (`Content-Length: 0` или отсутствие тела).

## На собеседовании

**Вопрос с собеседования:** «Почему при возврате 201 Created хорошей практикой является установка заголовка Location?»
**Ответ:** Заголовок `Location: /tasks/1` сообщает клиенту канонический URI вновь созданного ресурса, позволяя фронтенду или API-клиенту сразу перейти к созданному объекту без необходимости повторного поиска.
