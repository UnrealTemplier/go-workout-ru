# Полноценный in-memory CRUD REST API для сущности User с RWMutex

## Условие

Напиши **REST API** с полным CRUD (in-memory `map[int]User` + `sync.RWMutex`):
- `GET /users` — список
- `GET /users/{id}` — один пользователь
- `POST /users` — создать
- `PUT /users/{id}` — обновить
- `DELETE /users/{id}` — удалить

## Теория

CRUD (Create, Read, Update, Delete) — стандарт архитектуры REST:
- `GET` — идемпотентное получение ресурса (200 OK, 404 Not Found).
- `POST` — создание нового ресурса (201 Created с заголовком Location).
- `PUT` — полная замена ресурса по ID (200 OK).
- `DELETE` — удаление ресурса (204 No Content или 200 OK).
Для потокобезопасности хранилища в памяти используется `sync.RWMutex`: параллельные чтения `RLock` и эксклюзивная запись `Lock`.

## Решение по шагам

1. Опишем структуру `User { ID, Name, Email }`.
2. Создадим хранилище `UserStore` с `sync.RWMutex` и мапой `map[int]User`.
3. Зарегистрируем маршруты с новым синтаксисом Go 1.22+.
4. Протестируем весь жизненный цикл пользователя: Create -> List -> Get -> Update -> Delete.

## Код решения

**`main.go`**

```go
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"net/http/httptest"
	"strconv"
	"sync"
)

type User struct {
	ID    int    `json:"id"`
	Name  string `json:"name"`
	Email string `json:"email"`
}

type UserStore struct {
	mu     sync.RWMutex
	users  map[int]User
	nextID int
}

func NewUserStore() *UserStore {
	return &UserStore{users: make(map[int]User), nextID: 1}
}

func (s *UserStore) Create(u User) User {
	s.mu.Lock()
	defer s.mu.Unlock()
	u.ID = s.nextID
	s.nextID++
	s.users[u.ID] = u
	return u
}

func (s *UserStore) GetAll() []User {
	s.mu.RLock()
	defer s.mu.RUnlock()
	list := make([]User, 0, len(s.users))
	for _, u := range s.users {
		list = append(list, u)
	}
	return list
}

func (s *UserStore) Get(id int) (User, bool) {
	s.mu.RLock()
	defer s.mu.RUnlock()
	u, ok := s.users[id]
	return u, ok
}

func (s *UserStore) Update(id int, u User) bool {
	s.mu.Lock()
	defer s.mu.Unlock()
	if _, exists := s.users[id]; !exists {
		return false
	}
	u.ID = id
	s.users[id] = u
	return true
}

func (s *UserStore) Delete(id int) bool {
	s.mu.Lock()
	defer s.mu.Unlock()
	if _, exists := s.users[id]; !exists {
		return false
	}
	delete(s.users, id)
	return true
}

func main() {
	store := NewUserStore()
	mux := http.NewServeMux()

	mux.HandleFunc("GET /users", func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "application/json")
		_ = json.NewEncoder(w).Encode(store.GetAll())
	})

	mux.HandleFunc("POST /users", func(w http.ResponseWriter, r *http.Request) {
		var u User
		if err := json.NewDecoder(r.Body).Decode(&u); err != nil {
			http.Error(w, err.Error(), http.StatusBadRequest)
			return
		}
		created := store.Create(u)
		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusCreated)
		_ = json.NewEncoder(w).Encode(created)
	})

	mux.HandleFunc("GET /users/{id}", func(w http.ResponseWriter, r *http.Request) {
		id, err := strconv.Atoi(r.PathValue("id"))
		if err != nil {
			http.Error(w, "invalid id", http.StatusBadRequest)
			return
		}
		u, ok := store.Get(id)
		if !ok {
			http.NotFound(w, r)
			return
		}
		w.Header().Set("Content-Type", "application/json")
		_ = json.NewEncoder(w).Encode(u)
	})

	mux.HandleFunc("PUT /users/{id}", func(w http.ResponseWriter, r *http.Request) {
		id, err := strconv.Atoi(r.PathValue("id"))
		if err != nil {
			http.Error(w, "invalid id", http.StatusBadRequest)
			return
		}
		var u User
		if err := json.NewDecoder(r.Body).Decode(&u); err != nil {
			http.Error(w, err.Error(), http.StatusBadRequest)
			return
		}
		if !store.Update(id, u) {
			http.NotFound(w, r)
			return
		}
		w.WriteHeader(http.StatusOK)
		fmt.Fprint(w, `{"status":"updated"}`)
	})

	mux.HandleFunc("DELETE /users/{id}", func(w http.ResponseWriter, r *http.Request) {
		id, err := strconv.Atoi(r.PathValue("id"))
		if err != nil {
			http.Error(w, "invalid id", http.StatusBadRequest)
			return
		}
		if !store.Delete(id) {
			http.NotFound(w, r)
			return
		}
		w.WriteHeader(http.StatusNoContent)
	})

	ts := httptest.NewServer(mux)
	defer ts.Close()

	// 1. POST /users
	res, _ := http.Post(ts.URL+"/users", "application/json", bytes.NewBufferString(`{"name":"Иван","email":"ivan@ya.ru"}`))
	var created User
	_ = json.NewDecoder(res.Body).Decode(&created)
	res.Body.Close()
	fmt.Printf("[1. POST] ID: %d, Name: %s\n", created.ID, created.Name)

	// 2. GET /users/1
	res, _ = http.Get(ts.URL + "/users/1")
	b, _ := io.ReadAll(res.Body)
	res.Body.Close()
	fmt.Printf("[2. GET]  %s\n", string(b))

	// 3. DELETE /users/1
	req, _ := http.NewRequest(http.MethodDelete, ts.URL+"/users/1", nil)
	delRes, _ := http.DefaultClient.Do(req)
	delRes.Body.Close()
	fmt.Printf("[3. DEL]  Status: %d\n", delRes.StatusCode)
}
```

**`Терминал`**

```bash
go run main.go
# Вывод:
# [1. POST] ID: 1, Name: Иван
# [2. GET]  {"id":1,"name":"Иван","email":"ivan@ya.ru"}
# [3. DEL]  Status: 204
```

## Под капотом

В Go 1.22 роутер `ServeMux` компилирует методы (`GET`, `POST`, `DELETE`) и переменные `{id}` в дерево префиксов, избавляя от необходимости кастомных свитчей по `r.Method`.

## Подводные камни

Не забывайте, что чтение из мапы в Go при одновременной записи без мьютекса приводит к `fatal error: concurrent map read and map write`, роняющей весь сервер без возможности восстановления через `recover()`.

## На собеседовании

**Вопрос с собеседования:** «В чем отличие семантики PUT от PATCH в REST API?»
**Ответ:** `PUT` означает полную замену сущности (клиент обязан прислать все поля; неуказанные поля сбрасываются в значения по умолчанию). `PATCH` выполняет частичное обновление только тех полей, которые были явно переданы в теле запроса.
