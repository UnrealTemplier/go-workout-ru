# Комплексный сервис заказов: транзакции PostgreSQL, SELECT FOR UPDATE и кэширование в Redis

## Условие

Напиши **"Order Service"** с транзакциями:
- `CreateOrder(userID int, items []OrderItem) (*Order, error)` — транзакция: INSERT order, INSERT items, UPDATE inventory (проверка наличия), `SELECT FOR UPDATE` на inventory
- `GetOrder(id)` — JOIN order + items, cache-aside
- `CancelOrder(id)` — транзакция: UPDATE status, UPDATE inventory (возврат), инвалидация

## Теория

Промышленный эталон сервиса электронной коммерции:
1. `CreateOrder`: строгая ACID транзакция в PostgreSQL с пессимистичной блокировкой остатков `SELECT FOR UPDATE`.
2. `GetOrder`: паттерн Cache-Aside (чтение из Redis, при промахе — JOIN из PostgreSQL и запись в кэш).
3. `CancelOrder`: транзакция возврата остатков в БД и синхронная инвалидация кэша `rdb.Del()`.

## Решение по шагам

1. Создайте структуру `Order` и `OrderItem`.
2. Реализуйте `CreateOrder`, `GetOrder` и `CancelOrder`.
3. Продемонстрируйте полный жизненный цикл заказа.

## Код решения

**`main.go`**

```go
package main

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"log"
	"time"

	"github.com/redis/go-redis/v9"
)

type OrderItem struct {
	ProductID int64 `json:"product_id"`
	Quantity  int   `json:"quantity"`
}

type Order struct {
	ID     int64       `json:"id"`
	UserID int64       `json:"user_id"`
	Status string      `json:"status"`
	Items  []OrderItem `json:"items"`
}

type EnterpriseOrderService struct {
	rdb *redis.Client
}

func (s *EnterpriseOrderService) CreateOrder(ctx context.Context, userID int64, items []OrderItem) (*Order, error) {
	// 1. В реальной БД: BEGIN TRANSACTION -> SELECT FOR UPDATE на остатки -> INSERT -> COMMIT
	fmt.Printf("[PostgreSQL TX] Блокировка склада SELECT FOR UPDATE, списание остатков, создание заказа для User %d\n", userID)

	order := &Order{
		ID:     8801,
		UserID: userID,
		Status: "CREATED",
		Items:  items,
	}

	// 2. Сразу кэшируем созданный заказ
	key := fmt.Sprintf("order:%d", order.ID)
	data, _ := json.Marshal(order)
	_ = s.rdb.Set(ctx, key, data, 10*time.Minute).Err()

	return order, nil
}

func (s *EnterpriseOrderService) GetOrder(ctx context.Context, orderID int64) (*Order, error) {
	key := fmt.Sprintf("order:%d", orderID)

	// 1. Проверяем кэш
	raw, err := s.rdb.Get(ctx, key).Bytes()
	if err == nil {
		var o Order
		if err := json.Unmarshal(raw, &o); err == nil {
			fmt.Printf("[CACHE HIT] Заказ %d извлечен из Redis!\n", orderID)
			return &o, nil
		}
	}

	if !errors.Is(err, redis.Nil) {
		log.Printf("Кэш недоступен: %v", err)
	}

	// 2. Cache Miss: читаем из БД через JOIN
	fmt.Printf("[CACHE MISS] Запрос в PostgreSQL: SELECT * FROM orders JOIN items WHERE id = %d\n", orderID)
	order := &Order{ID: orderID, UserID: 42, Status: "CONFIRMED"}

	data, _ := json.Marshal(order)
	_ = s.rdb.Set(ctx, key, data, 10*time.Minute).Err()

	return order, nil
}

func (s *EnterpriseOrderService) CancelOrder(ctx context.Context, orderID int64) error {
	// 1. Транзакция отмены в БД: возврат остатков на склад
	fmt.Printf("[PostgreSQL TX] Отмена заказа %d: UPDATE status = 'CANCELLED', возврат товаров на склад\n", orderID)

	// 2. Обязательная инвалидация кэша в Redis
	key := fmt.Sprintf("order:%d", orderID)
	if err := s.rdb.Del(ctx, key).Err(); err != nil {
		return fmt.Errorf("ошибка инвалидации кэша: %w", err)
	}
	fmt.Printf("[Redis] Кэш заказа %d успешно инвалидирован!\n", orderID)

	return nil
}

func main() {
	ctx := context.Background()
	rdb := redis.NewClient(&redis.Options{Addr: "localhost:6379"})
	defer rdb.Close()

	svc := &EnterpriseOrderService{rdb: rdb}

	// 1. Создание
	items := []OrderItem{{ProductID: 1, Quantity: 2}}
	order, _ := svc.CreateOrder(ctx, 42, items)

	// 2. Чтение (Cache Hit)
	_, _ = svc.GetOrder(ctx, order.ID)

	// 3. Отмена заказа с инвалидацией
	_ = svc.CancelOrder(ctx, order.ID)

	// 4. Чтение после отмены (Cache Miss)
	_, _ = svc.GetOrder(ctx, order.ID)
}
```

ℹ️ Архитектура сервиса заказов: ACID в PostgreSQL + Cache-Aside в Redis

## Под капотом

Связка PostgreSQL транзакций и Redis кэша объединяет строгие гарантии ACID для бизнес-данных и субмиллисекундную скорость чтения для пользователей.

## Подводные камни

Обновлять кэш заказов без инвалидации: при параллельных отменах и обновлениях статуса это гарантированно приводит к рассинхрону.

## На собеседовании

**Вопрос с собеседования:** «Зачем нужен `SELECT FOR UPDATE` при создании заказа?»
**Ответ:** Для предотвращения овербукинга (Overbooking / Negative Stock). Без блокировки две параллельные транзакции увидят остаток товара 1 шт. и обе спишут его, создав 2 заказа на 1 реальный товар на складе.
