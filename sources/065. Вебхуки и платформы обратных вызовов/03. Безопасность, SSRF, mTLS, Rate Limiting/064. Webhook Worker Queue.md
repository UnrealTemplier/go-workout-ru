## Теория

При реализации очереди задач доставки на PostgreSQL критически важно исключить конкуренцию воркеров за строки.
Каноничная схема промышленной очереди:
```sql
CREATE TABLE webhook_delivery_queue (
    id BIGSERIAL PRIMARY KEY,
    webhook_id VARCHAR(64) NOT NULL,
    payload JSONB NOT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'PENDING',
    attempt_count INT NOT NULL DEFAULT 0,
    next_retry_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    locked_by VARCHAR(64),
    locked_at TIMESTAMPTZ
);
```
Паттерн атомарного захвата пачки задач:
```sql
WITH ready_tasks AS (
    SELECT id FROM webhook_delivery_queue
    WHERE status IN ('PENDING', 'RETRYING')
      AND next_retry_at <= NOW()
    ORDER BY next_retry_at ASC
    LIMIT 20
    FOR UPDATE SKIP LOCKED
)
UPDATE webhook_delivery_queue
SET status = 'PROCESSING', locked_by = $1, locked_at = NOW()
WHERE id IN (SELECT id FROM ready_tasks)
RETURNING id, webhook_id, payload, attempt_count;
```
Этот запрос обеспечивает параллельную работу сотен воркеров без взаимных блокировок.

## Решение по шагам

1. Опишите модель задачи `QueueTask`.
2. Реализуйте сервис очереди с эмуляцией выборки `FOR UPDATE SKIP LOCKED`.
3. Запустите 3 параллельных воркера и отправьте 10 задач в очередь.
4. Продемонстрируйте, что каждая задача обрабатывается строго одним воркером без дублирования.

## Код решения

**`postgres_skip_locked_queue.go`**

```go
package main

import (
	"fmt"
	"sync"
	"time"
)

type QueueTask struct {
	ID        int
	WebhookID string
	Status    string // PENDING, PROCESSING, COMPLETED
	LockedBy  string
}

type SafePostgresQueue struct {
	mu    sync.Mutex
	tasks []*QueueTask
}

func (q *SafePostgresQueue) Enqueue(webhookID string) {
	q.mu.Lock()
	defer q.mu.Unlock()
	id := len(q.tasks) + 1
	q.tasks = append(q.tasks, &QueueTask{ID: id, WebhookID: webhookID, Status: "PENDING"})
}

// FetchBatch эмулирует SELECT ... FOR UPDATE SKIP LOCKED
func (q *SafePostgresQueue) FetchBatch(workerName string, limit int) []*QueueTask {
	q.mu.Lock()
	defer q.mu.Unlock()

	var batch []*QueueTask
	for _, t := range q.tasks {
		if t.Status == "PENDING" {
			t.Status = "PROCESSING"
			t.LockedBy = workerName
			batch = append(batch, t)
			if len(batch) >= limit {
				break
			}
		}
	}
	return batch
}

func (q *SafePostgresQueue) Complete(taskID int) {
	q.mu.Lock()
	defer q.mu.Unlock()
	for _, t := range q.tasks {
		if t.ID == taskID {
			t.Status = "COMPLETED"
			return
		}
	}
}

func main() {
	queue := &SafePostgresQueue{}
	for i := 1; i <= 9; i++ {
		queue.Enqueue(fmt.Sprintf("hook_sub_%d", i))
	}

	var wg sync.WaitGroup
	startWorker := func(name string) {
		defer wg.Done()
		tasks := queue.FetchBatch(name, 3)
		for _, t := range tasks {
			fmt.Printf("[%s] Executed task #%d for %s\n", name, t.ID, t.WebhookID)
			time.Sleep(10 * time.Millisecond)
			queue.Complete(t.ID)
		}
	}

	wg.Add(3)
	go startWorker("Worker-Alpha")
	go startWorker("Worker-Beta")
	go startWorker("Worker-Gamma")

	wg.Wait()
	fmt.Println("All 9 queue tasks processed concurrently without duplicates.")
}
```

## Под капотом

В PostgreSQL конструкция `FOR UPDATE SKIP LOCKED` при сканировании B-Tree индекса проверяет статус транзакционного замка строки в памяти `tuple lock`. Если замок установлен другим воркером, транзакция не блокируется в ожидании (как при обычном `FOR UPDATE`), а мгновенно переходит к следующей подходящей строке. Это устраняет деградацию throughput при увеличении числа параллельных воркеров.

## Подводные камни

1. Зависшие задачи в статусе `PROCESSING` (Worker Crash): если воркер умер во время доставки, задача останется в статусе `PROCESSING` навсегда. Необходим фоновый процесс `Sweeper`, сбрасывающий задачи с `locked_at < NOW() - INTERVAL '5 minutes'` обратно в `PENDING`.
2. Выборка по 1 задаче: создает избыточный сетевой оверхед к БД. Задачи должны забираться пачками (батчами по 20–100 штук).
3. Отсутствие индекса по `(status, next_retry_at)`.

## На собеседовании

**Вопрос (Avito/PostgreSQL):** Как предотвратить накопление зависших "мертвых" блокировок (Dead Locks / Orphaned Tasks) в очереди на базе PostgreSQL, если один из подов воркеров был убит по OOMKilled?
**Ответ:** Используется паттерн **Heartbeat Lock / Lock Expiration**:
Каждый воркер при захвате задачи выставляет `locked_at = NOW()`. Пока воркер жив, он периодически обновляет `locked_at = NOW()`. Отдельный фоновый таймер раз в 1 минуту выполняет восстановительный запрос:
```sql
UPDATE webhook_delivery_queue
SET status = 'RETRYING', locked_by = NULL
WHERE status = 'PROCESSING' AND locked_at < NOW() - INTERVAL '3 minutes';
```
Это возвращает брошенные задачи в очередь повторов без ручного вмешательства инженеров.
