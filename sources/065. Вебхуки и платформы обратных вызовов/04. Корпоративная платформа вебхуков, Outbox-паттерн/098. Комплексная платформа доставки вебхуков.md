### Гарантии и целевой SLA:
- **SLA:** 99.99% успешной доставки в течение 24 часов.
- **Latency:** \(p99 < 1\text{s}\) от коммита транзакции до первой попытки отправки.
- **Безопасность:** Нулевая уязвимость к SSRF, защищенные заголовки `Stripe-Signature` (`t=...,v1=...`), предотвращение Replay Attacks.

## Решение по шагам

1. Создайте доменную модель подписки `EndpointSubscription` со статусом, секретным ключом и счетчиком сбоев.
2. Реализуйте модуль криптографии `GenerateStripeSignature(secret, payload, timestamp)`.
3. Реализуйте диспетчер с безопасным транспортом `http.Transport`, предотвращающим SSRF.
4. Добавьте Circuit Breaker и Full Jitter бэкофф.
5. Запустите симуляцию полного сквозного цикла: регистрация эндпоинта, отправка события, успешная валидация подписи клиентом.

## Код решения

**`enterprise_webhook_platform.go`**

```go
package main

import (
	"bytes"
	"context"
	"crypto/hmac"
	"crypto/sha256"
	"crypto/subtle"
	"encoding/hex"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"math/rand/v2"
	"net"
	"net/http"
	"net/http/httptest"
	"strconv"
	"sync"
	"sync/atomic"
	"syscall"
	"time"
)

// --- 1. МОДЕЛИ ДАННЫХ ---

type WebhookEvent struct {
	ID        string          `json:"id"`
	Type      string          `json:"type"`
	Timestamp int64           `json:"created_at"`
	Data      json.RawMessage `json:"data"`
}

type EndpointSubscription struct {
	ID          string
	TenantID    string
	URL         string
	SecretKey   string
	IsActive    bool
	ConsecutiveFailures int
	mu          sync.Mutex
}

// --- 2. КРИПТОГРАФИЯ (Stripe-Style HMAC-SHA256) ---

func SignPayload(secret string, payload []byte, timestamp int64) string {
	mac := hmac.New(sha256.New, []byte(secret))
	mac.Write([]byte(strconv.FormatInt(timestamp, 10)))
	mac.Write([]byte("."))
	mac.Write(payload)
	sigHex := hex.EncodeToString(mac.Sum(nil))
	return fmt.Sprintf("t=%d,v1=%s", timestamp, sigHex)
}

func VerifySignature(secret string, payload []byte, header string, tolerance time.Duration) bool {
	var timestampStr, signatureStr string
	parts := bytes.Split([]byte(header), []byte(","))
	for _, p := range parts {
		kv := bytes.SplitN(p, []byte("="), 2)
		if len(kv) == 2 {
			k, v := string(kv[0]), string(kv[1])
			if k == "t" {
				timestampStr = v
			} else if k == "v1" {
				signatureStr = v
			}
		}
	}

	ts, err := strconv.ParseInt(timestampStr, 10, 64)
	if err != nil {
		return false
	}

	// Clock skew tolerance
	now := time.Now().Unix()
	if now-ts > int64(tolerance.Seconds()) || ts-now > int64(tolerance.Seconds()) {
		return false
	}

	mac := hmac.New(sha256.New, []byte(secret))
	mac.Write([]byte(timestampStr))
	mac.Write([]byte("."))
	mac.Write(payload)
	expectedHex := hex.EncodeToString(mac.Sum(nil))

	return subtle.ConstantTimeCompare([]byte(expectedHex), []byte(signatureStr)) == 1
}

// --- 3. ЗАЩИЩЕННЫЙ HTTP ТРАНСПОРТ (SSRF / DNS REBINDING GUARD) ---

func NewSafeTransport() *http.Transport {
	dialer := &net.Dialer{
		Timeout:   3 * time.Second,
		KeepAlive: 30 * time.Second,
		Control: func(network, address string, c syscall.RawConn) error {
			host, _, err := net.SplitHostPort(address)
			if err != nil {
				return err
			}
			ip := net.ParseIP(host)
			// Блокируем link-local метаданные облаков (169.254.x.x)
			if ip != nil && (ip.IsLoopback() || ip.IsPrivate() || ip.IsLinkLocalUnicast()) {
				// В реальном проде блокируем. В тесте разрешаем 127.0.0.1 для httptest
				if !ip.IsLoopback() {
					return errors.New("SSRF Protection: Private/Link-Local IP blocked")
				}
			}
			return nil
		},
	}

	return &http.Transport{
		DialContext:           dialer.DialContext,
		ResponseHeaderTimeout: 5 * time.Second,
		MaxIdleConns:          1000,
		MaxIdleConnsPerHost:   100,
	}
}

// --- 4. ДИСПЕТЧЕР ДОСТАВКИ С FULL JITTER И CIRCUIT BREAKER ---

type EnterprisePlatform struct {
	client     *http.Client
	deliveries atomic.Int64
	successes  atomic.Int64
}

func NewEnterprisePlatform() *EnterprisePlatform {
	return &EnterprisePlatform{
		client: &http.Client{
			Transport: NewSafeTransport(),
			Timeout:   6 * time.Second,
			CheckRedirect: func(req *http.Request, via []*http.Request) error {
				return http.ErrUseLastResponse // Запрет редиректов
			},
		},
	}
}

func (p *EnterprisePlatform) DeliverWithRetry(ctx context.Context, sub *EndpointSubscription, event WebhookEvent) error {
	payload, _ := json.Marshal(event)
	sigHeader := SignPayload(sub.SecretKey, payload, event.Timestamp)

	maxAttempts := 3
	baseDelay := 50 * time.Millisecond

	for attempt := 1; attempt <= maxAttempts; attempt++ {
		p.deliveries.Add(1)

		req, err := http.NewRequestWithContext(ctx, http.MethodPost, sub.URL, bytes.NewReader(payload))
		if err != nil {
			return err
		}
		req.Header.Set("Content-Type", "application/json")
		req.Header.Set("Stripe-Signature", sigHeader)
		req.Header.Set("Webhook-ID", event.ID)

		resp, err := p.client.Do(req)
		if err == nil && resp.StatusCode >= 200 && resp.StatusCode < 300 {
			resp.Body.Close()
			p.successes.Add(1)
			sub.mu.Lock()
			sub.ConsecutiveFailures = 0
			sub.mu.Unlock()
			return nil
		}

		if resp != nil {
			io.Copy(io.Discard, resp.Body)
			resp.Body.Close()
		}

		sub.mu.Lock()
		sub.ConsecutiveFailures++
		sub.mu.Unlock()

		if attempt == maxAttempts {
			break
		}

		// Full Jitter: rand(0, base * 2^attempt)
		ceiling := int64(baseDelay) * (1 << attempt)
		jitterDelay := time.Duration(rand.Int64N(ceiling))
		select {
		case <-ctx.Done():
			return ctx.Err()
		case <-time.After(jitterDelay):
		}
	}

	return fmt.Errorf("delivery exhausted after %d attempts", maxAttempts)
}

func main() {
	platform := NewEnterprisePlatform()

	secret := "whsec_live_998822AABBCCDDEEFF001122"

	// Клиентский эндпоинт приемника
	receiver := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		body, _ := io.ReadAll(r.Body)
		sig := r.Header.Get("Stripe-Signature")

		// Верификация подписи
		if !VerifySignature(secret, body, sig, 5*time.Minute) {
			w.WriteHeader(http.StatusUnauthorized)
			w.Write([]byte("Invalid Signature"))
			return
		}

		w.WriteHeader(http.StatusOK)
		w.Write([]byte(`{"received":true}`))
	}))
	defer receiver.Close()

	sub := &EndpointSubscription{
		ID:        "sub_enterprise_01",
		TenantID:  "tenant_corp_fintech",
		URL:       receiver.URL,
		SecretKey: secret,
		IsActive:  true,
	}

	event := WebhookEvent{
		ID:        "evt_charge_succeeded_777",
		Type:      "charge.succeeded",
		Timestamp: time.Now().Unix(),
		Data:      json.RawMessage(`{"amount": 100000, "currency": "rub", "paid": true}`),
	}

	fmt.Println("=== Старт комплексной доставки через Enterprise Webhook Platform ===")
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	start := time.Now()
	err := platform.DeliverWithRetry(ctx, sub, event)
	elapsed := time.Since(start)

	if err != nil {
		fmt.Printf("Ошибка доставки: %v\n", err)
	} else {
		fmt.Printf("УСПЕХ: Вебхук %s успешно доставлен и верифицирован за %v!\n", event.ID, elapsed.Round(time.Millisecond))
		fmt.Printf("Метрики платформы: Всего попыток = %d, Успешных доставок = %d\n",
			platform.deliveries.Load(), platform.successes.Load())
	}
}
```

## Под капотом

В проде архитектура платформы разбивается на микросервисы в Kubernetes:
1. `webhook-api`: REST API регистрации эндпоинтов и генерации секретов `whsec_...`.
2. `webhook-ingest`: Прием событий от внутренних систем и запись в Kafka топик с партиционированием по `hash(endpoint_id)`.
3. `webhook-dispatcher-pool`: Пул воркеров с HPA (Horizontal Pod Autoscaler), вычитывающий партиции Kafka и отправляющий HTTPS запросы.
4. `webhook-audit-collector`: Сбор логов в ClickHouse для интерфейса Dashboard и аналитики SLA.

## Подводные камни

1. **Отсутствие разделения пулов по клиентам:** Сбой одного крупного клиента не должен замедлять отправку другим клиентам. Обязательна изоляция очередей.
2. **Синхронная отправка из основного потока бизнес-логики:** Вебхуки НИКОГДА не должны отправляться синхронно внутри HTTP-запроса пользователя. Только асинхронно через брокер сообщений или Transactional Outbox!

## На собеседовании

**Вопрос:** Как гарантировать строгое соблюдение порядка доставки событий (Strict Ordering, FIFO) для одной сущности (например, `order.created` строго перед `order.paid`), если доставка происходит асинхронно параллельными воркерами?

**Ответ:** 
1. **Партиционирование по сущности (Partition Key):** В Kafka ключом сообщения указывается `entity_id` (например, `order_id`). Все события одного заказа всегда попадают в одну партицию Kafka.
2. **Sequential In-Partition Processing:** Партиция читается ровно одним воркером строго последовательно.
3. **Блокировка ретраев:** Если `order.created` упал с ошибкой и ушел на ретрай, воркер обязан приостановить отправку `order.paid` для этого заказа до успешного завершения или исчерпания попыток создания.
