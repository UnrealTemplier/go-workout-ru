## Теория

Финальный синтез современных облачных микросервисов на Go в корпоративной среде: 1) `Service Mesh (Data Plane)`: прозрачный mTLS, автоматические ретраи с джиттером, Outlier Detection предохранители; 2) `Прикладной рантайм (Go)`: строгая обработка `ctx.Done()`, отсутствие дублирующих ретраев (No Retry Amplification), проброс контекста трассировки (W3C `traceparent` и B3) через `http.RoundTripper`, проверка идемпотентности транзакций по `Idempotency-Key` и авторизация по SPIFFE ID из `XFCC`. Этот сплав превращает распределенную систему в несокрушимый HighLoad-сервис мирового уровня.

```go
package main

import (
	"context"
	"encoding/json"
	"fmt"
	"log"
	"net/http"
	"os"
	"os/signal"
	"strings"
	"sync"
	"syscall"
	"time"
)

// MasterOrderService объединяет Service Mesh, mTLS, трассировку и идемпотентность.
type MasterOrderService struct {
	mu          sync.Mutex
	idempotency map[string]string
}

func (s *MasterOrderService) ProcessOrder(ctx context.Context, orderID, key string) (string, error) {
	s.mu.Lock()
	if cached, exists := s.idempotency[key]; exists {
		s.mu.Unlock()
		log.Printf("[MasterService] Idempotency cache hit for key %s", key)
		return cached, nil
	}
	s.mu.Unlock()

	// Имитация работы с проверкой отмены контекста
	select {
	case <-ctx.Done():
		return "", ctx.Err()
	case <-time.After(50 * time.Millisecond):
	}

	result := fmt.Sprintf(`{"status":"success","order_id":"%s"}`, orderID)
	s.mu.Lock()
	s.idempotency[key] = result
	s.mu.Unlock()

	return result, nil
}

func main() {
	svc := &MasterOrderService{idempotency: make(map[string]string)}

	mux := http.NewServeMux()
	mux.HandleFunc("/api/v1/orders", func(w http.ResponseWriter, r *http.Request) {
		// 1. W3C Trace Context
		traceParent := r.Header.Get("traceparent")
		if traceParent != "" {
			w.Header().Set("traceparent", traceParent)
		}

		// 2. SPIFFE Identity Verification
		xfcc := r.Header.Get("X-Forwarded-Client-Cert")
		if xfcc != "" && !strings.Contains(xfcc, "cluster.local") {
			http.Error(w, "Forbidden: Untrusted SPIFFE realm", http.StatusForbidden)
			return
		}

		// 3. Idempotent Processing
		key := r.Header.Get("Idempotency-Key")
		if key == "" {
			key = "default-idem-key"
		}

		res, err := svc.ProcessOrder(r.Context(), "ord-9901", key)
		if err != nil {
			http.Error(w, "Cancelled or failed", http.StatusRequestTimeout)
			return
		}

		w.Header().Set("Content-Type", "application/json")
		w.WriteHeader(http.StatusOK)
		w.Write([]byte(res))
	})

	srv := &http.Server{
		Addr:         ":8080",
		Handler:      mux,
		ReadTimeout:  5 * time.Second,
		WriteTimeout: 10 * time.Second,
	}

	stop := make(chan os.Signal, 1)
	signal.Notify(stop, syscall.SIGINT, syscall.SIGTERM)

	go func() {
		log.Println("Enterprise Service Mesh Go Master Service running on :8080")
		_ = srv.ListenAndServe()
	}()

	<-stop
	log.Println("Gracefully stopping service...")
	shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	_ = srv.Shutdown(shutdownCtx)
	fmt.Println("Capstone Master Service cleanly terminated")
}
```
