# Ручная реализация безопасной композиции AES-CBC с HMAC (Encrypt-then-MAC)

## Условие

**AES-CBC с HMAC (Encrypt-then-MAC).**: Реализуйте шифрование через AES-CBC вручную (с PKCS#7 padding), а затем вычислите HMAC-SHA256 от шифротекста. Ключи для AES и HMAC должны быть разными (выведите их из мастер-ключа через HKDF). Проверьте порядок: сначала проверяйте HMAC, потом расшифровывайте.

## Теория

До появления стандартизированных режимов AEAD классическим способом построения стойкого шифра была схема **Encrypt-then-MAC**:
1. Открытый текст дополняется PKCS#7 паддингом и шифруется в режиме AES-CBC с помощью ключа шифрования \(K_{enc}\).
2. Полученный шифротекст (включая вектор IV) подписывается кодом аутентификации сообщения HMAC-SHA256 с помощью ключа аутентификации \(K_{mac}\).
3. **Критическое правило разделения ключей**: ключи \(K_{enc}\) и \(K_{mac}\) обязаны быть криптографически независимыми! Использование одного ключа и для AES, и для HMAC является грубой уязвимостью. Для их порождения из одного мастер-секрета используется функция **HKDF** (HMAC-based Extract-and-Expand Key Derivation Function — RFC 5869).
4. При дешифрации HMAC обязан проверяться в константном времени *до* вызова дешифратора CBC.

## Решение по шагам

1. Возьмите общий мастер-ключ (32 байта).
2. С помощью `golang.org/x/crypto/hkdf` выведите два независимых 32-байтных ключа: `encKey` с контекстом `info="enc"` и `macKey` с контекстом `info="mac"`.
3. Зашифруйте данные в режиме AES-CBC с добавлением PKCS#7 паддинга.
4. Вычислите HMAC-SHA256 от среза `[IV || Ciphertext]`.
5. Соберите пакет: `[IV 16 байт][Ciphertext ...][HMAC 32 байта]`.
6. При дешифрации сначала проверьте HMAC через `hmac.Equal`, и только при успехе снимите шифрование CBC и удалите паддинг.

## Код решения

**`encrypt_then_mac.go`**

```go
package main

import (
	"bytes"
	"crypto/aes"
	"crypto/cipher"
	"crypto/hmac"
	"crypto/rand"
	"crypto/sha256"
	"errors"
	"fmt"
	"io"

	"golang.org/x/crypto/hkdf"
)

// DeriveKeys выводит два независимых ключа с помощью HKDF-SHA256
func DeriveKeys(masterKey []byte) (encKey, macKey []byte, err error) {
	encKey = make([]byte, 32)
	macKey = make([]byte, 32)

	// HKDF Expand для ключа шифрования
	kdfEnc := hkdf.New(sha256.New, masterKey, nil, []byte("AES-CBC-Encryption-Key"))
	if _, err = io.ReadFull(kdfEnc, encKey); err != nil {
		return nil, nil, err
	}

	// HKDF Expand для ключа аутентификации
	kdfMac := hkdf.New(sha256.New, masterKey, nil, []byte("HMAC-SHA256-Auth-Key"))
	if _, err = io.ReadFull(kdfMac, macKey); err != nil {
		return nil, nil, err
	}

	return encKey, macKey, nil
}

func pkcs7Pad(data []byte, blockSize int) []byte {
	padding := blockSize - len(data)%blockSize
	padtext := bytes.Repeat([]byte{byte(padding)}, padding)
	return append(data, padtext...)
}

func pkcs7Unpad(data []byte, blockSize int) ([]byte, error) {
	if len(data) == 0 || len(data)%blockSize != 0 {
		return nil, errors.New("invalid block size")
	}
	padLen := int(data[len(data)-1])
	if padLen == 0 || padLen > blockSize {
		return nil, errors.New("invalid padding len")
	}
	for i := len(data) - padLen; i < len(data); i++ {
		if data[i] != byte(padLen) {
			return nil, errors.New("corrupt padding")
		}
	}
	return data[:len(data)-padLen], nil
}

// EncryptThenMAC шифрует и подписывает пакет
func EncryptThenMAC(plaintext, masterKey []byte) ([]byte, error) {
	encKey, macKey, err := DeriveKeys(masterKey)
	if err != nil {
		return nil, err
	}

	block, err := aes.NewCipher(encKey)
	if err != nil {
		return nil, err
	}

	padded := pkcs7Pad(plaintext, aes.BlockSize)
	// [IV 16 байт][Ciphertext ...][HMAC 32 байта]
	totalLen := aes.BlockSize + len(padded) + sha256.Size
	packet := make([]byte, totalLen)

	iv := packet[:aes.BlockSize]
	if _, err := io.ReadFull(rand.Reader, iv); err != nil {
		return nil, err
	}

	ciphertext := packet[aes.BlockSize : aes.BlockSize+len(padded)]
	mode := cipher.NewCBCEncrypter(block, iv)
	mode.CryptBlocks(ciphertext, padded)

	// Вычисляем HMAC от [IV || Ciphertext]
	mac := hmac.New(sha256.New, macKey)
	mac.Write(packet[:aes.BlockSize+len(padded)])
	tag := mac.Sum(nil)

	copy(packet[aes.BlockSize+len(padded):], tag)
	return packet, nil
}

// DecryptThenVerify проверяет HMAC и только потом расшифровывает
func DecryptThenVerify(packet, masterKey []byte) ([]byte, error) {
	minLen := aes.BlockSize + aes.BlockSize + sha256.Size
	if len(packet) < minLen {
		return nil, errors.New("packet too short")
	}

	encKey, macKey, err := DeriveKeys(masterKey)
	if err != nil {
		return nil, err
	}

	payloadLen := len(packet) - sha256.Size
	encryptedData := packet[:payloadLen]
	receivedTag := packet[payloadLen:]

	// 1. ПЕРВЫМ ДЕЛОМ ПРОВЕРЯЕМ HMAC В КОНСТАНТНОМ ВРЕМЕНИ!
	mac := hmac.New(sha256.New, macKey)
	mac.Write(encryptedData)
	expectedTag := mac.Sum(nil)

	if !hmac.Equal(receivedTag, expectedTag) {
		return nil, errors.New("HMAC verification failed: tampering detected")
	}

	// 2. Только после успешной проверки HMAC расшифровываем CBC
	iv := encryptedData[:aes.BlockSize]
	ciphertext := encryptedData[aes.BlockSize:]

	block, err := aes.NewCipher(encKey)
	if err != nil {
		return nil, err
	}

	decryptedPadded := make([]byte, len(ciphertext))
	mode := cipher.NewCBCDecrypter(block, iv)
	mode.CryptBlocks(decryptedPadded, ciphertext)

	return pkcs7Unpad(decryptedPadded, aes.BlockSize)
}

func main() {
	masterKey := make([]byte, 32)
	_ = rand.Read(masterKey)

	msg := []byte("Safe Encrypt-then-MAC Architecture with HKDF Key Separation")

	packet, err := EncryptThenMAC(msg, masterKey)
	if err != nil {
		panic(err)
	}
	fmt.Printf("Пакет успешно сформирован: %d байт\n", len(packet))

	decrypted, err := DecryptThenVerify(packet, masterKey)
	if err != nil {
		panic(err)
	}
	fmt.Printf("Успешно расшифровано: %s\n", string(decrypted))
}
```

## Под капотом

Функция HKDF (RFC 5869) состоит из двух фаз: `Extract` и `Expand`.
Фаза Extract сжимает энтропию в псевдослучайный ключ \(PRK = HMAC-Hash(Salt, IKM)\).
Фаза Expand вырабатывает криптографически независимые ключи произвольной длины путем конкатенации раундов HMAC с контекстной строкой `info`. Даже если ключ шифрования будет скомпрометирован, математически невозможно восстановить ключ HMAC.

## Подводные камни

Вычисление HMAC только от шифротекста без включения IV: это позволяет злоумышленнику подменить IV на произвольный вектор, исказив первый блок открытого текста.

## На собеседовании

**Вопрос с собеседования:** «Почему в схеме Encrypt-then-MAC вектор инициализации IV обязательно должен входить в данные, подписываемые HMAC?»
**Ответ:** Дешифрование первого блока в CBC вычисляется как \(P_1 = D_K(C_1) \oplus IV\). Если IV не защищен HMAC, злоумышленник в канале связи может изменить байты IV (\(IV' = IV \oplus \Delta\)). Проверка HMAC пройдет успешно, так как шифротекст \(C\) не тронут, но в расшифрованном первом блоке значение изменится на \(P_1 \oplus \Delta\). Включение IV в HMAC предотвращает эту атаку.
