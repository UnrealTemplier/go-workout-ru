# Генерация ключей RSA и сохранение приватного ключа с парольной защитой

## Условие

**Генерация пары ключей RSA.**: Сгенерируйте 2048-битный RSA-ключ, сохраните приватный ключ в PEM-файл с паролем, публичный — в открытый PEM. Прочитайте ключи обратно.

## Теория

При экспорте приватных ключей на постоянные носители критически важно защитить их парольной фразой (Passphrase Protection), чтобы утечка файла не привела к автоматической компрометации всей системы.

В современном стандарте **PKCS#8 v2 (RFC 5208 / RFC 5958)** для шифрования приватных ключей используется:
1. **KDF**: PBKDF2 или scrypt для вывода ключа из пароля пользователя с солью и большим числом итераций.
2. **Cipher**: AES-256-CBC или AES-256-GCM.
3. В стандартной библиотеке Go публичный ключ сохраняется в формате **PKIX (`x509.MarshalPKIXPublicKey`)**, а приватный — в **PKCS#8 (`x509.MarshalPKCS8PrivateKey`)**.
4. Для обратной загрузки открытого ключа используется `x509.ParsePKIXPublicKey` с приведением к `*rsa.PublicKey`.

## Решение по шагам

1. Сгенерируйте пару RSA-2048 через `rsa.GenerateKey(rand.Reader, 2048)`.
2. Экспортируйте публичный ключ в PEM `PUBLIC KEY` через `x509.MarshalPKIXPublicKey`.
3. Зашифруйте приватный ключ паролем: выведите ключ AES-256 через `pbkdf2` и зашифруйте DER-байты через AES-GCM.
4. Оберните зашифрованные данные в PEM блок `ENCRYPTED PRIVATE KEY`.
5. Реализуйте функцию обратной загрузки с проверкой пароля.

## Код решения

**`rsa_pem_pass.go`**

```go
package main

import (
	"crypto/aes"
	"crypto/cipher"
	"crypto/rand"
	"crypto/rsa"
	"crypto/sha256"
	"crypto/x509"
	"encoding/pem"
	"errors"
	"fmt"
	"io"
	"os"

	"golang.org/x/crypto/pbkdf2"
)

// ExportPublicKey сохраняет открытый ключ в стандартном формате PKIX PEM
func ExportPublicKey(pubKey *rsa.PublicKey, path string) error {
	derBytes, err := x509.MarshalPKIXPublicKey(pubKey)
	if err != nil {
		return err
	}
	block := &pem.Block{
		Type:  "PUBLIC KEY",
		Bytes: derBytes,
	}
	return os.WriteFile(path, pem.EncodeToMemory(block), 0644)
}

// ExportEncryptedPrivateKey шифрует PKCS#8 приватный ключ паролем
func ExportEncryptedPrivateKey(privKey *rsa.PrivateKey, password string, path string) error {
	derBytes, err := x509.MarshalPKCS8PrivateKey(privKey)
	if err != nil {
		return err
	}

	salt := make([]byte, 16)
	_, _ = io.ReadFull(rand.Reader, salt)

	// Выводим ключ AES-256 из пароля через PBKDF2
	aesKey := pbkdf2.Key([]byte(password), salt, 100_000, 32, sha256.New)

	block, _ := aes.NewCipher(aesKey)
	gcm, _ := cipher.NewGCM(block)
	nonce := make([]byte, gcm.NonceSize())
	_, _ = io.ReadFull(rand.Reader, nonce)

	ciphertext := gcm.Seal(nil, nonce, derBytes, nil)

	// Упаковываем: [Salt 16B][Nonce 12B][Ciphertext]
	payload := append(salt, nonce...)
	payload = append(payload, ciphertext...)

	pemBlock := &pem.Block{
		Type:  "ENCRYPTED PRIVATE KEY",
		Bytes: payload,
	}

	return os.WriteFile(path, pem.EncodeToMemory(pemBlock), 0600)
}

// LoadEncryptedPrivateKey загружает и расшифровывает приватный ключ
func LoadEncryptedPrivateKey(path string, password string) (*rsa.PrivateKey, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return nil, err
	}

	block, _ := pem.Decode(data)
	if block == nil || block.Type != "ENCRYPTED PRIVATE KEY" {
		return nil, errors.New("invalid PEM block type")
	}

	payload := block.Bytes
	if len(payload) < 28 {
		return nil, errors.New("payload too short")
	}

	salt := payload[:16]
	nonce := payload[16:28]
	ciphertext := payload[28:]

	aesKey := pbkdf2.Key([]byte(password), salt, 100_000, 32, sha256.New)
	aesBlock, _ := aes.NewCipher(aesKey)
	gcm, _ := cipher.NewGCM(aesBlock)

	derBytes, err := gcm.Open(nil, nonce, ciphertext, nil)
	if err != nil {
		return nil, errors.New("failed to decrypt private key: incorrect password or corrupt data")
	}

	parsedKey, err := x509.ParsePKCS8PrivateKey(derBytes)
	if err != nil {
		return nil, err
	}

	rsaKey, ok := parsedKey.(*rsa.PrivateKey)
	if !ok {
		return nil, errors.New("not an RSA private key")
	}

	return rsaKey, nil
}

func main() {
	privKey, _ := rsa.GenerateKey(rand.Reader, 2048)

	pubFile := "server_public.pem"
	privFile := "server_private.pem"
	defer os.Remove(pubFile)
	defer os.Remove(privFile)

	_ = ExportPublicKey(&privKey.PublicKey, pubFile)
	_ = ExportEncryptedPrivateKey(privKey, "SuperSafeMasterPass#2026", privFile)
	fmt.Println("Ключи RSA успешно экспортированы (приватный зашифрован с PBKDF2/AES-GCM)!")

	// Чтение с правильным паролем
	loadedKey, err := LoadEncryptedPrivateKey(privFile, "SuperSafeMasterPass#2026")
	if err != nil {
		panic(err)
	}
	fmt.Printf("Приватный ключ успешно расшифрован! N.BitLen = %d\n", loadedKey.N.BitLen())

	// Попытка чтения с неверным паролем
	_, err = LoadEncryptedPrivateKey(privFile, "WrongPassword!")
	if err != nil {
		fmt.Printf("Ожидаемый отказ при неверном пароле: %v\n", err)
	}
}
```

## Под капотом

Стандарт `x509.MarshalPKIXPublicKey` сериализует открытый ключ в структуру SubjectPublicKeyInfo (RFC 5280): OID алгоритма `rsaEncryption` (1.2.840.113549.1.1.1) и битовую строку с телом ключа `RSAPublicKey` (модуль \(n\) и открытая экспонента \(e\)).

## Подводные камни

Использование дефолтного `x509.EncryptPEMBlock` из стандартной библиотеки (помечен как deprecated, так как использует небезопасный MD5 и CBC режим).

## На собеседовании

**Вопрос с собеседования:** «В чем разница между форматами `BEGIN RSA PRIVATE KEY` и `BEGIN PRIVATE KEY`?»
**Ответ:** `BEGIN RSA PRIVATE KEY` — это старый формат PKCS#1, хранящий исключительно сырые параметры RSA (модуль \(n\), экспоненты \(e, d\), простые числа \(p, q\)). Он не содержит метаданных об алгоритме и не поддерживает эллиптические кривые. `BEGIN PRIVATE KEY` — это универсальный стандарт PKCS#8, где перед телом ключа идет структура `AlgorithmIdentifier` с ASN.1 OID алгоритма, что позволяет в одном формате хранить любые ключи (RSA, ECDSA, Ed25519, Kyber).
