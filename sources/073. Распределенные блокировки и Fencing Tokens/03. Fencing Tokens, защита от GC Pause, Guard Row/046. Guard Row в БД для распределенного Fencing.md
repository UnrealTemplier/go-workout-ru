# Guard Row в БД для распределенного Fencing без внешнего координатора

## Теория

В распределенных системах добавление внешнего сервиса координации (etcd, Consul, ZooKeeper) часто увеличивает эксплуатационную сложность инфраструктуры. Если ваша система уже использует реляционную СУБД с поддержкой строгих транзакций ACID (PostgreSQL или MySQL в режиме SERIALIZABLE / READ COMMITTED), распределенный мьютекс и Fencing Token можно реализовать через паттерн **Guard Row** прямо в базе данных.

Схема паттерна:
1. Создается служебная таблица `locks (resource_id PRIMARY KEY, owner_id TEXT, token BIGINT, expires_at TIMESTAMPTZ)`.
2. Захват блокировки выполняется атомарным SQL-запросом `INSERT ... ON CONFLICT DO UPDATE`:
   - Если записи нет: создается строка с `token = 1`, `expires_at = NOW() + TTL`.
   - Если запись есть: блокировка продлевается или перехватывается, только если текущий срок истек (`WHERE locks.expires_at < NOW() OR locks.owner_id = EXCLUDED.owner_id`), при этом токен инкрементируется: `SET token = locks.token + 1`.
3. Любая бизнес-операция (например, `UPDATE accounts SET balance = balance - 100 WHERE id = 1 AND fencing_token = :token`) проверяет совпадение токена.

**Почему это гарантирует безопасность даже при задержках и GC-паузах?**
База данных является единственным источником истины (Single Source of Truth). Если у воркера случилась STW GC-пауза на 1 минуту, блокировка в таблице `locks` истечет, ее перехватит другой воркер и токен возрастет (например, с 5 до 6). Когда зомби-воркер очнется с токеном 5 и выполнит `UPDATE ... WHERE fencing_token = 5`, запрос затронет 0 строк (`rows_affected == 0`), предотвратив порчу данных.

## Решение по шагам

1. Опишите таблицу `distributed_locks` и целевую таблицу бизнес-данных.
2. Реализуйте метод `AcquireGuardLock(db, resourceID, ownerID, ttl) (token int64, err error)`.
3. Сформируйте атомарный SQL запрос с `ON CONFLICT (resource_id) DO UPDATE ... WHERE expires_at < NOW() ... RETURNING token`.
4. Реализуйте метод выполнения бизнес-операции, обязательно проверяющий `token` в блоке `WHERE`.
5. Продемонстрируйте сценарий отторжения записи от зомби-воркера.

## Код решения

**`main.go`**

```go
package main

import (
	"context"
	"database/sql"
	"errors"
	"fmt"
	"time"
)

// SQL DDL (для PostgreSQL):
// CREATE TABLE distributed_locks (
//     resource_id VARCHAR(128) PRIMARY KEY,
//     owner_id VARCHAR(128) NOT NULL,
//     token BIGINT NOT NULL DEFAULT 1,
//     expires_at TIMESTAMP WITH TIME ZONE NOT NULL
// );
//
// CREATE TABLE accounts (
//     id VARCHAR(64) PRIMARY KEY,
//     balance NUMERIC NOT NULL,
//     fencing_token BIGINT NOT NULL DEFAULT 0
// );

type MockDBClient struct {
	// Симуляция таблиц реляционной БД в памяти с семантикой блокировок
	lockResource string
	lockOwner    string
	lockToken    int64
	lockExpires  time.Time

	accountBalance int64
	accountToken   int64
}

func (db *MockDBClient) AcquireGuardLock(ctx context.Context, resID, ownerID string, ttl time.Duration) (int64, error) {
	now := time.Now()
	// Имитация атомарного SQL:
	// INSERT INTO distributed_locks (resource_id, owner_id, token, expires_at)
	// VALUES ($1, $2, 1, NOW() + $3)
	// ON CONFLICT (resource_id) DO UPDATE
	// SET token = distributed_locks.token + 1,
	//     owner_id = EXCLUDED.owner_id,
	//     expires_at = EXCLUDED.expires_at
	// WHERE distributed_locks.expires_at < NOW() OR distributed_locks.owner_id = EXCLUDED.owner_id
	// RETURNING token;

	if db.lockResource == "" {
		db.lockResource = resID
		db.lockOwner = ownerID
		db.lockToken = 1
		db.lockExpires = now.Add(ttl)
		return db.lockToken, nil
	}

	if db.lockExpires.Before(now) || db.lockOwner == ownerID {
		db.lockToken++
		db.lockOwner = ownerID
		db.lockExpires = now.Add(ttl)
		return db.lockToken, nil
	}

	return 0, errors.New("lock is currently held by active owner")
}

func (db *MockDBClient) UpdateBalanceWithFencing(ctx context.Context, delta int64, token int64) error {
	// Имитация SQL:
	// UPDATE accounts SET balance = balance + $1, fencing_token = $2
	// WHERE id = 'acc-1' AND fencing_token < $2;
	if token <= db.accountToken {
		return fmt.Errorf("fencing violation: provided token %d <= current db token %d (zombie write rejected)",
			token, db.accountToken)
	}

	db.accountBalance += delta
	db.accountToken = token
	return nil
}

func main() {
	db := &MockDBClient{
		accountBalance: 1000,
		accountToken:   0,
	}

	ctx := context.Background()

	// 1. Воркер-1 берет блокировку на 100мс
	token1, err := db.AcquireGuardLock(ctx, "account-lock", "worker-1", 100*time.Millisecond)
	if err != nil {
		panic(err)
	}
	fmt.Printf("[Worker-1] Acquired lock with token %d\n", token1)

	// 2. Воркер-1 уходит в симуляцию долгой GC-паузы (200мс)
	fmt.Println("[Worker-1] Enters long STW GC pause (sleeping 200ms)...")
	time.Sleep(200 * time.Millisecond)

	// 3. За это время Воркер-2 видит истекший срок, перехватывает блокировку
	token2, err := db.AcquireGuardLock(ctx, "account-lock", "worker-2", 500*time.Millisecond)
	if err != nil {
		panic(err)
	}
	fmt.Printf("[Worker-2] Successfully acquired expired lock! New token: %d\n", token2)

	// Воркер-2 успешно обновляет баланс
	if err := db.UpdateBalanceWithFencing(ctx, -200, token2); err != nil {
		fmt.Println("[Worker-2] Error:", err)
	} else {
		fmt.Printf("[Worker-2] Updated balance. New balance: %d, token: %d\n", db.accountBalance, db.accountToken)
	}

	// 4. Воркер-1 очнулся от GC-паузы и пытается применить списание со своим старым token1
	fmt.Printf("[Worker-1] Woke up! Trying to update balance with stale token %d...\n", token1)
	err = db.UpdateBalanceWithFencing(ctx, -100, token1)
	if err != nil {
		fmt.Printf("[Worker-1] WRITE REJECTED BY DB: %v\n", err)
	}

	fmt.Printf("Final Database State: Balance=%d, FencingToken=%d\n", db.accountBalance, db.accountToken)
	_ = sql.ErrNoRows
}
```

ℹ️ Реализация Guard Row в БД с атомарным получением монотонного токена и проверкой в WHERE

## Под капотом

В СУБД вроде PostgreSQL атомарный запрос `INSERT ... ON CONFLICT DO UPDATE ... RETURNING` использует строгие спинлоки и страничный локинг кортежей на диске. Если несколько транзакций конкурируют за одну строку в `distributed_locks`, они выстраиваются в очередь по `XMAX` (идентификатор транзакции блокировки строки). Предикат `WHERE distributed_locks.expires_at < NOW()` гарантирует, что чужая активная блокировка не будет перехвачена. А инкремент `token = token + 1` гарантирует строгую монотонность без необходимости в отдельной последовательности `SEQUENCE`.

## Подводные камни

1. Рассинхронизация времени (`clock skew`) между серверами приложений и сервером СУБД: всегда используйте `NOW()` на стороне базы данных, а не время клиента.
2. Пропуск условия `WHERE fencing_token < :token` в бизнес-запросах: если забыть это условие, зомби-воркер все равно перезапишет данные.

## На собеседовании

«Почему Guard Row в PostgreSQL предпочтительнее внешнего ZooKeeper/etcd для монолитов или умеренно нагруженных микросервисов?»
Ответ: Guard Row устраняет распределенный консенсус между двумя разными системами (координатором и БД). В классической схеме с etcd может произойти сбой между освобождением лока и коммитом в БД. В случае Guard Row блокировка и проверка токена могут выполняться в единой транзакции БД, гарантируя строгую согласованность при нулевых затратах на поддержку отдельного кластера координации.
