## Теория

Блок 23 завершает фундаментальный цикл проектирования распределенных систем. Паттерны Transactional Outbox и Transactional Inbox представляют собой высшую ступень надежности обмена данными между микросервисами:
1) Transactional Outbox решает проблему Dual Write: бизнес-сущность и исходящие события сохраняются в одной ACID-транзакции СУБД. Ни одно событие не теряется при сбоях сети;
2) Change Data Capture (Debezium) вычитывает WAL/oplog базы данных без нагрузки на таблицы;
3) Apache Kafka обеспечивает масштабируемую упорядоченную буферизацию сообщений;
4) Transactional Inbox с ограничением UNIQUE на `message_id` устраняет дубликаты At-Least-Once доставки;
5) Идемпотентные обработчики и Saga Orchestration завершают цепочку, обеспечивая сквозную надежность микросервисов в условиях сетевых сбоев, рестартов подов и высокой конкуренции.

## Код решения

**`outbox_inbox_full_stack.go`**

```go
package main

import (
	"context"
	"fmt"
	"sync"
	"time"
)

// Архитектурный синтез всех компонентов
type SystemBlueprint struct {
	OutboxTable []string
	KafkaTopic  []string
	InboxTable  map[string]bool
	mu          sync.Mutex
}

func NewSystemBlueprint() *SystemBlueprint {
	return &SystemBlueprint{InboxTable: make(map[string]bool)}
}

// 1. Service A: Локальная транзакция + Outbox
func (s *SystemBlueprint) ServiceA_CreateOrder(orderID string) {
	s.mu.Lock()
	defer s.mu.Unlock()

	fmt.Printf("[Service A (Postgres)] BEGIN TX: Inserting order %s + Outbox Event\n", orderID)
	event := fmt.Sprintf("evt-order-created-%s", orderID)
	s.OutboxTable = append(s.OutboxTable, event)
	fmt.Printf("[Service A (Postgres)] COMMIT TX: Event %s safely stored in Outbox.\n", event)
}

// 2. CDC / Debezium Poller: Перенос из Outbox в Kafka
func (s *SystemBlueprint) CDC_RelayEvents() {
	s.mu.Lock()
	defer s.mu.Unlock()

	for _, evt := range s.OutboxTable {
		fmt.Printf("[CDC Engine] Reading WAL -> Streaming %s to Kafka topic 'order-events'\n", evt)
		s.KafkaTopic = append(s.KafkaTopic, evt)
	}
	s.OutboxTable = nil // Помечено как обработанное
}

// 3. Service B: Inbox Consumer + Бизнес-логика
func (s *SystemBlueprint) ServiceB_ConsumeEvents() {
	s.mu.Lock()
	defer s.mu.Unlock()

	for _, msg := range s.KafkaTopic {
		fmt.Printf("[Service B] Consumer received: %s\n", msg)
		if s.InboxTable[msg] {
			fmt.Printf("[Service B INBOX] Duplicate %s detected! Committing offset without duplicate processing.\n", msg)
			continue
		}

		// Атомарная обработка в БД Service B
		s.InboxTable[msg] = true
		fmt.Printf("[Service B (Postgres)] BEGIN TX: Executing billing charge + INSERT into Inbox -> COMMIT TX OK\n")
	}
}

func main() {
	sys := NewSystemBlueprint()
	_ = context.Background()

	fmt.Println("=== FULL-STACK DISTRIBUTED TRANSACTIONS RUNTIME ===")

	// 1. Сервис А создает заказ
	sys.ServiceA_CreateOrder("ord-7001")

	// 2. Debezium передает в Kafka
	time.Sleep(20 * time.Millisecond)
	sys.CDC_RelayEvents()

	// 3. Сервис Б потребляет сообщение
	time.Sleep(20 * time.Millisecond)
	sys.ServiceB_ConsumeEvents()

	// 4. Симуляция дубликата из-за рестарта консьюмера Kafka
	fmt.Println("\n--- Simulating Kafka Redelivery / Duplicate Event ---")
	sys.ServiceB_ConsumeEvents()

	fmt.Println("\n>>> Architecture validation completed: Exactly-Once business semantics achieved! <<<")
}
```

```bash
# Инструменты и переменные окружения для локальной тренировки
export KAFKA_BOOTSTRAP_SERVERS="localhost:9092"
export POSTGRES_OUTBOX_URL="postgres://postgres:secret@localhost:5432/orders_db?sslmode=disable"
export DEBEZIUM_CONNECT_URL="http://localhost:8083/connectors"

# 1. Запуск инфраструктуры (Kafka, Debezium, PostgreSQL)
docker-compose -f docker-compose.distributed.yml up -d

# 2. Регистрация Debezium Outbox коннектора
curl -i -X POST -H "Accept:application/json" -H "Content-Type:application/json" \
  $DEBEZIUM_CONNECT_URL \
  -d '{
    "name": "orders-outbox-connector",
    "config": {
      "connector.class": "io.debezium.connector.postgresql.PostgresConnector",
      "tasks.max": "1",
      "plugin.name": "pgoutput",
      "database.hostname": "postgres",
      "database.port": "5432",
      "database.user": "postgres",
      "database.password": "secret",
      "database.dbname": "orders_db",
      "database.server.name": "dbserver1",
      "table.include.list": "public.outbox_events",
      "tombstones.on.delete": "false"
    }
  }'

# 3. Мониторинг топика Kafka в реальном времени
kafka-console-consumer --bootstrap-server $KAFKA_BOOTSTRAP_SERVERS \
  --topic dbserver1.public.outbox_events \
  --from-beginning
```

## Под капотом

Связка Outbox + CDC + Kafka + Inbox превращает ненадежную сеть в детерминированный конвейер данных. Даже при падении любого компонента (БД сервиса А, брокера Kafka, сервиса Б) данные остаются в безопасности. После рестарта система автоматически догоняет лаг, гарантируя абсолютную надежность корпоративного уровня.

## Подводные камни

1. Забытый мониторинг лага: накопление терабайтов в Outbox при поломке CDC положит диск PostgreSQL.
2. Очистка Outbox без подтверждения в Kafka: удалять строки из Outbox можно только после того, как оффсет в Kafka закоммичен всеми репликами (acks=all).
