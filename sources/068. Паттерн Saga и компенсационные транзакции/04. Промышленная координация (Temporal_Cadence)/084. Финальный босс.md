## Теория

Платежный шлюз уровня Stripe обрабатывает финансовые транзакции с нулевой терпимостью к дубликатам. Архитектура строится на двухуровневой защите идемпотентности: 1) Fast Path: Redis `SET key req_hash NX EX 86400` захватывает эксклюзивный лок на ключ на время выполнения; 2) Persistent Path: реляционная СУБД PostgreSQL с `INSERT INTO idempotency_records (...) ON CONFLICT (key) DO NOTHING`. Если повторный запрос приходит во время исполнения первого, он получает HTTP `409 Conflict` (или блокируется до получения ответа). Если первый запрос завершен, сервер мгновенно отдает сохраненный HTTP статус и JSON-ответ из базы данных без обращения к платежному ядру.

## Решение по шагам

## Код решения

**`stripe_idempotency_gateway.go`**

```go
package main

import (
	"context"
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"fmt"
	"sync"
	"time"
)

type CachedResponse struct {
	StatusCode int
	Body       string
}

type GatewayStore struct {
	mu          sync.Mutex
	locks       map[string]time.Time      // Redis Fast Path Mock (SET NX EX)
	records     map[string]CachedResponse // Postgres Persistent Storage Mock
	chargesData map[string]float64
}

func NewGatewayStore() *GatewayStore {
	return &GatewayStore{
		locks:       make(map[string]time.Time),
		records:     make(map[string]CachedResponse),
		chargesData: make(map[string]float64),
	}
}

func (s *GatewayStore) AcquireOrGet(key, payloadHash string) (*CachedResponse, bool, error) {
	s.mu.Lock()
	defer s.mu.Unlock()

	// 1. Проверяем готовый ответ в постоянном хранилище (Postgres)
	if resp, exists := s.records[key]; exists {
		return &resp, false, nil
	}

	// 2. Проверяем блокировку (In-flight запрос)
	if lockTime, locked := s.locks[key]; locked {
		if time.Since(lockTime) < 5*time.Second {
			return nil, false, errors.New("409 Conflict: Concurrent request in progress")
		}
	}

	// Захватываем лок
	s.locks[key] = time.Now()
	return nil, true, nil
}

func (s *GatewayStore) CommitResponse(key string, chargeID string, amount float64, resp CachedResponse) {
	s.mu.Lock()
	defer s.mu.Unlock()

	delete(s.locks, key)
	s.records[key] = resp
	if chargeID != "" {
		s.chargesData[chargeID] = amount
	}
}

type StripePaymentGateway struct {
	store *GatewayStore
}

func (g *StripePaymentGateway) Charge(ctx context.Context, idempKey string, amount float64) (CachedResponse, error) {
	hash := hashPayload(fmt.Sprintf("%.2f", amount))

	cached, acquired, err := g.store.AcquireOrGet(idempKey, hash)
	if err != nil {
		return CachedResponse{}, err
	}
	if !acquired {
		fmt.Printf("[Gateway Cache HIT] Key %s served from persistent cache!\n", idempKey)
		return *cached, nil
	}

	// Выполнение реального процессинга
	fmt.Printf("[Gateway Processing] Charging $%.2f for key %s...\n", amount, idempKey)
	time.Sleep(30 * time.Millisecond) // Эмуляция вызова банковского ядра
	chargeID := fmt.Sprintf("ch_%d", time.Now().UnixNano())

	resp := CachedResponse{
		StatusCode: 200,
		Body:       fmt.Sprintf("{\"id\":\"%s\",\"amount\":%.2f,\"status\":\"succeeded\"}", chargeID, amount),
	}

	g.store.CommitResponse(idempKey, chargeID, amount, resp)
	fmt.Printf("[Gateway Success] Charge %s committed to storage.\n", chargeID)
	return resp, nil
}

func hashPayload(payload string) string {
	h := sha256.Sum256([]byte(payload))
	return hex.EncodeToString(h[:])
}

func main() {
	store := NewGatewayStore()
	gateway := &StripePaymentGateway{store: store}
	ctx := context.Background()

	idempKey := "idem-key-user-invoice-404"

	// Запрос 1: Первичная авторизация
	fmt.Println("=== REQUEST 1: Primary Charge ===")
	resp1, err1 := gateway.Charge(ctx, idempKey, 99.99)
	if err1 != nil {
		fmt.Printf("Error: %v\n", err1)
	} else {
		fmt.Printf("Status: %d, Body: %s\n", resp1.StatusCode, resp1.Body)
	}

	// Запрос 2: Повторный запрос от клиента (клиент подумал, что был сетевой сбой)
	fmt.Println("\n=== REQUEST 2: Replay with Identical Key ===")
	resp2, err2 := gateway.Charge(ctx, idempKey, 99.99)
	if err2 != nil {
		fmt.Printf("Error: %v\n", err2)
	} else {
		fmt.Printf("Status: %d, Body: %s\n", resp2.StatusCode, resp2.Body)
	}
}
```

## Под капотом

В архитектуре Stripe Idempotency Layer сохраняет не только полезную нагрузку ответа, но и HTTP статус-код, а также оригинальные заголовки ответа (`Content-Type`, `Request-Id`). Хеш тела входящего запроса сохраняется в БД. Если клиент присылает тот же ключ, но с другим телом (другая сумма или валюта), шлюз отклоняет запрос ошибкой `422 Unprocessable Entity: Payload mismatch`.

## Подводные камни

1. Забытый лок при падении воркера: если воркер взял лок в Redis и упал с паникой до фиксации в Postgres, ключ окажется заблокирован на весь срок TTL. Необходим `defer` с освобождением лока при ошибке.
2. Утечка данных между арендаторами: Idempotency Key обязан быть изолирован по `AccountID / API-Key`.

## На собеседовании

Вопрос: Зачем проверять хеш тела запроса вместе с ключом идемпотентности?
Ответ: Чтобы предотвратить критическую ошибку подмены контекста (Key Collisions / Replay Attack). Если злоумышленник перехватит чужой ключ идемпотентности и отправит его со своим номером счета, без проверки хеша запроса сервер вернул бы успешный кэшированный ответ, породив уязвимость или выполнив ошибочное действие.
