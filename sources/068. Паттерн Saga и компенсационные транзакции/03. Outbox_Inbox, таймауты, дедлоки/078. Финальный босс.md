## Теория

Финальный босс объединяет все изученные архитектурные концепции в единую высоконадежную систему: 1. HTTP API Order Service принимает заказ с проверкой уникальности `Idempotency-Key` (RFC 9457); 2. В единой ACID-транзакции PostgreSQL заказ сохраняется со статусом `PENDING`, а событие инициации саги записывается в локальную таблицу `outbox_events`; 3. Saga Execution Coordinator (SEC) управляет шагами распределенного процесса: резервирование оплаты (Payment Service), резервирование складских остатков (Warehouse Service) и создание накладной доставки (Shipping Service); 4. При возникновении ошибки на любом шаге оркестратор запускает компенсирующие транзакции в строго обратном порядке (LIFO), возвращая систему в согласованное состояние; 5. Все внешние вызовы снабжены таймаутами контекста и передают Idempotency Key.

## Решение по шагам

1. Реализуйте хранилище заказов и Outbox в единой транзакции с проверкой ключа идемпотентности.
2. Опишите интерфейсы сервисов `PaymentService`, `WarehouseService`, `ShippingService` с компенсациями.
3. Реализуйте структуру `OrderSagaOrchestrator` со стеком компенсаций LIFO.
4. Смоделируйте сценарий успешного заказа (Happy Path).
5. Смоделируйте сценарий сбоя на этапе доставки (Shipping) и подтвердите откат склада и возврата денег.

## Код решения

**`ecommerce_saga_platform.go`**

```go
package main

import (
	"context"
	"errors"
	"fmt"
	"sync"
	"time"
)

// Модели домена
type Order struct {
	ID             string
	CustomerID     string
	Amount         float64
	Status         string
	IdempotencyKey string
}

type OutboxEvent struct {
	EventID   string
	OrderID   string
	EventType string
	Payload   string
}

// OrderRepository с поддержкой транзакционного Outbox
type OrderRepository struct {
	mu           sync.Mutex
	orders       map[string]Order
	idempMap     map[string]string // IdempotencyKey -> OrderID
	outboxEvents []OutboxEvent
}

func NewOrderRepository() *OrderRepository {
	return &OrderRepository{
		orders:   make(map[string]Order),
		idempMap: make(map[string]string),
	}
}

func (r *OrderRepository) CreateOrderAtomic(ctx context.Context, o Order) (string, bool, error) {
	r.mu.Lock()
	defer r.mu.Unlock()

	// Проверка идемпотентности
	if existingID, exists := r.idempMap[o.IdempotencyKey]; exists {
		fmt.Printf("[Order DB] Duplicate request with IdempotencyKey %s -> Returning existing Order %s\n", o.IdempotencyKey, existingID)
		return existingID, true, nil
	}

	// Атомарная запись заказа и события в Outbox
	r.orders[o.ID] = o
	r.idempMap[o.IdempotencyKey] = o.ID
	r.outboxEvents = append(r.outboxEvents, OutboxEvent{
		EventID:   fmt.Sprintf("evt-%s", o.ID),
		OrderID:   o.ID,
		EventType: "OrderCreated",
		Payload:   fmt.Sprintf("{\"amount\": %.2f}", o.Amount),
	})

	fmt.Printf("[Order DB TX] Order %s and OutboxEvent committed atomically.\n", o.ID)
	return o.ID, false, nil
}

func (r *OrderRepository) UpdateStatus(orderID, status string) {
	r.mu.Lock()
	defer r.mu.Unlock()
	if ord, ok := r.orders[orderID]; ok {
		ord.Status = status
		r.orders[orderID] = ord
		fmt.Printf("[Order DB] Order %s status updated to: %s\n", orderID, status)
	}
}

// Внешние сервисы платформы
type PlatformServices struct {
	simulatedFailure string
}

func (s *PlatformServices) ChargePayment(ctx context.Context, orderID string, amount float64) error {
	fmt.Printf("[Payment Service] Charging $%.2f for %s -> OK\n", amount, orderID)
	return nil
}

func (s *PlatformServices) RefundPayment(ctx context.Context, orderID string) error {
	fmt.Printf("[Payment Service COMPENSATION] Refunded payment for %s -> OK\n", orderID)
	return nil
}

func (s *PlatformServices) ReserveWarehouse(ctx context.Context, orderID string) error {
	fmt.Printf("[Warehouse Service] Reserved inventory for %s -> OK\n", orderID)
	return nil
}

func (s *PlatformServices) ReleaseWarehouse(ctx context.Context, orderID string) error {
	fmt.Printf("[Warehouse Service COMPENSATION] Released warehouse items for %s -> OK\n", orderID)
	return nil
}

func (s *PlatformServices) CreateShipping(ctx context.Context, orderID string) error {
	if s.simulatedFailure == "shipping" {
		fmt.Printf("[Shipping Service] Delivery booking failed: Couriers unavailable!\n")
		return errors.New("courier fleet unavailable")
	}
	fmt.Printf("[Shipping Service] Created shipping label for %s -> OK\n", orderID)
	return nil
}

// Оркестратор распределенной Саги
type ECommerceSagaCoordinator struct {
	orderRepo *OrderRepository
	services  *PlatformServices
}

func (c *ECommerceSagaCoordinator) ProcessOrder(ctx context.Context, order Order) error {
	orderID, isDuplicate, err := c.orderRepo.CreateOrderAtomic(ctx, order)
	if err != nil {
		return err
	}
	if isDuplicate {
		return nil
	}

	var compensations []func()

	// Шаг 1: Оплата
	if err := c.services.ChargePayment(ctx, orderID, order.Amount); err != nil {
		c.orderRepo.UpdateStatus(orderID, "REJECTED_PAYMENT_FAILED")
		return err
	}
	compensations = append(compensations, func() {
		_ = c.services.RefundPayment(context.Background(), orderID)
	})

	// Шаг 2: Склад
	if err := c.services.ReserveWarehouse(ctx, orderID); err != nil {
		c.orderRepo.UpdateStatus(orderID, "REJECTED_OUT_OF_STOCK")
		c.rollback(compensations)
		return err
	}
	compensations = append(compensations, func() {
		_ = c.services.ReleaseWarehouse(context.Background(), orderID)
	})

	// Шаг 3: Доставка
	if err := c.services.CreateShipping(ctx, orderID); err != nil {
		c.orderRepo.UpdateStatus(orderID, "FAILED_SHIPPING_UNAVAILABLE")
		c.rollback(compensations)
		return err
	}

	c.orderRepo.UpdateStatus(orderID, "COMPLETED")
	fmt.Printf("[Saga Coordinator] Order %s successfully fulfilled!\n\n", orderID)
	return nil
}

func (c *ECommerceSagaCoordinator) rollback(compensations []func()) {
	fmt.Printf("[Saga Coordinator] Triggering %d compensations in LIFO order...\n", len(compensations))
	for i := len(compensations) - 1; i >= 0; i-- {
		compensations[i]()
	}
}

func main() {
	repo := NewOrderRepository()
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()

	// Сценарий 1: Успешный заказ (Happy Path)
	fmt.Println("=== SCENARIO 1: Happy Path Order ===")
	servicesHappy := &PlatformServices{}
	coordHappy := &ECommerceSagaCoordinator{orderRepo: repo, services: servicesHappy}

	order1 := Order{
		ID:             "order-101",
		CustomerID:     "cust-alice",
		Amount:         850.0,
		Status:         "PENDING",
		IdempotencyKey: "idemp-key-alice-101",
	}
	_ = coordHappy.ProcessOrder(ctx, order1)

	// Сценарий 2: Повторный запрос с тем же Idempotency Key
	fmt.Println("=== SCENARIO 2: Idempotency Key Replay ===")
	_ = coordHappy.ProcessOrder(ctx, order1)

	// Сценарий 3: Сбой на этапе доставки с полным откатом склада и оплаты
	fmt.Println("\n=== SCENARIO 3: Shipping Failure & LIFO Compensation ===")
	servicesFailure := &PlatformServices{simulatedFailure: "shipping"}
	coordFailure := &ECommerceSagaCoordinator{orderRepo: repo, services: servicesFailure}

	order2 := Order{
		ID:             "order-102",
		CustomerID:     "cust-bob",
		Amount:         1420.0,
		Status:         "PENDING",
		IdempotencyKey: "idemp-key-bob-102",
	}
	_ = coordFailure.ProcessOrder(ctx, order2)
}
```

## Под капотом

Архитектура финального босса реализует паттерн Orchestration with Transactional Outbox. Клиент получает быстрый HTTP-ответ 202 Accepted сразу после записи в `orders` и `outbox_events`. Далее асинхронный координатор читает событие и запускает распределенный пайплайн. Использование стека LIFO (Last-In-First-Out) для компенсаций математически доказано как единственно верный порядок отката: последнее успешно выполненное действие должно компенсироваться первым, чтобы не нарушать временные инварианты взаимосвязанных ресурсов.

## Подводные камни

1. Потеря контекста транзакции при асинхронном запуске: если забыть передать корреляционные заголовки в микросервисы, распределенная трассировка распадется на несвязанные спаны.
2. Отсутствие таймаутов на компенсирующих вызовах: если сервис склада зависнет во время компенсации, весь откат остановится. Для каждого шага компенсации обязан задаваться строгий `context.WithTimeout`.

## На собеседовании

Вопрос: Каковы главные компромиссы (trade-offs) использования Саги вместо классического 2PC в HighLoad системах?
Ответ: Главный выигрыш Саги — высокая доступность и горизонтальная масштабируемость (AP/BASE вместо CP/ACID). Нет долгих распределенных блокировок ресурсов БД на время сетевых раунд-трипов. Главная цена Саги — потеря свойства изоляции (Isolation в ACID): промежуточные результаты шагов видны другим транзакциям (Dirty Reads). Для минимизации этого риска применяются семантические блокировки (Semantic Locks) и версионирование остатков.
