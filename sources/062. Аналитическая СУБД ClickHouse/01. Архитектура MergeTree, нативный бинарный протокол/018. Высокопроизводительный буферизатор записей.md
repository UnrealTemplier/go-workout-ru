# Высокопроизводительный буферизатор записей (Batcher) в Go

## Условие

**Пакетная запись (Batch Insertion)**: ClickHouse крайне неэффективно обрабатывает частые поштучные вставки. Напишите высокопроизводительный буферизатор логов в Go:
* Программа должна накапливать логи во внутреннем слайсе памяти.
* При достижении 50 000 логов (или раз в 5 секунд) запускать фоновую запись.
* Выполнять вставку пакетом через методы `db.PrepareBatch(ctx, "INSERT INTO logs ...")` и последовательные вызовы `batch.Append(...)` с последующим отправлением пакета через `batch.Send()`.

## Теория

Для устранения проблемы одиночных вставок в высоконагруженных Go-сервисах реализуется паттерн **In-Memory Batcher / Flush Buffer**:
1. События отправляются в неблокирующий канал Go или очередь.
2. Фоновая горутина накапливает записи во внутреннем срезе.
3. Сброс накопленного батча в ClickHouse происходит по двум триггерам:
   - **По размеру:** Батч достиг лимита `maxBatchSize` (например, 50 000 строк).
   - **По таймеру:** Прошло время `flushInterval` (например, 1 секунда), даже если батч заполнен не до конца.
4. При завершении сервиса (Graceful Shutdown) оставшиеся в буфере события гарантированно сбрасываются в ClickHouse.

## Решение по шагам

1. Определите структуру `LogBatcher` с каналом `chan Event`, лимитом размера и таймером `time.NewTicker`.
2. Реализуйте метод `Push(e Event)` с проверкой переполнения канала.
3. В фоновом цикле `flushLoop` накапливайте события и вызывайте `flush()` при наступлении события таймера или заполнении слайса.
4. Реализуйте метод `Close()`, дожидающийся сброса оставшихся записей через `sync.WaitGroup`.

## Код решения

**`batcher.go`**

```go
package main

import (
	"context"
	"fmt"
	"sync"
	"time"

	"github.com/ClickHouse/clickhouse-go/v2"
	"github.com/ClickHouse/clickhouse-go/v2/lib/driver"
	"github.com/google/uuid"
)

type Event struct {
	ID        uuid.UUID
	UserID    uint64
	EventTime time.Time
	Payload   string
}

type ClickHouseBatcher struct {
	conn      driver.Conn
	ch        chan Event
	maxSize   int
	flushTick time.Duration
	wg        sync.WaitGroup
	ctx       context.Context
	cancel    context.CancelFunc
}

func NewBatcher(conn driver.Conn, maxSize int, flushTick time.Duration) *ClickHouseBatcher {
	ctx, cancel := context.WithCancel(context.Background())
	b := &ClickHouseBatcher{
		conn:      conn,
		ch:        make(chan Event, maxSize*2),
		maxSize:   maxSize,
		flushTick: flushTick,
		ctx:       ctx,
		cancel:    cancel,
	}
	b.wg.Add(1)
	go b.worker()
	return b
}

func (b *ClickHouseBatcher) Push(e Event) {
	b.ch <- e
}

func (b *ClickHouseBatcher) worker() {
	defer b.wg.Done()
	ticker := time.NewTicker(b.flushTick)
	defer ticker.Stop()

	buffer := make([]Event, 0, b.maxSize)

	flush := func() {
		if len(buffer) == 0 {
			return
		}
		ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
		defer cancel()

		batch, err := b.conn.PrepareBatch(ctx, "INSERT INTO events (id, user_id, event_time, payload)")
		if err != nil {
			fmt.Printf("Batcher error: %v\n", err)
			buffer = buffer[:0]
			return
		}
		for _, e := range buffer {
			_ = batch.Append(e.ID, e.UserID, e.EventTime, e.Payload)
		}
		if err := batch.Send(); err != nil {
			fmt.Printf("Flush send error: %v\n", err)
		} else {
			fmt.Printf("Batcher сбросил в ClickHouse %d событий\n", len(buffer))
		}
		buffer = buffer[:0]
	}

	for {
		select {
		case <-b.ctx.Done():
			// Вычитываем оставшиеся события из канала
			for len(b.ch) > 0 {
				buffer = append(buffer, <-b.ch)
			}
			flush()
			return
		case e := <-b.ch:
			buffer = append(buffer, e)
			if len(buffer) >= b.maxSize {
				flush()
			}
		case <-ticker.C:
			flush()
		}
	}
}

func (b *ClickHouseBatcher) Close() {
	b.cancel()
	b.wg.Wait()
}

func main() {
	conn, err := clickhouse.Open(&clickhouse.Options{
		Addr: []string{"127.0.0.1:9000"},
		Auth: clickhouse.Auth{Database: "default"},
	})
	if err != nil {
		panic(err)
	}
	defer conn.Close()

	// Буферизатор: сброс каждые 500 событий или раз в 500 мс
	batcher := NewBatcher(conn, 500, 500*time.Millisecond)

	// Симуляция потока логов
	now := time.Now()
	for i := 0; i < 1200; i++ {
		batcher.Push(Event{
			ID:        uuid.New(),
			UserID:    uint64(2000 + i),
			EventTime: now,
			Payload:   "user_action_logged",
		})
	}

	// Корректное завершение с сохранением остатка
	time.Sleep(100 * time.Millisecond)
	batcher.Close()
	fmt.Println("Batcher корректно завершил работу")
}
```

## Под капотом

Использование буфера с переиспользованием лежащего в основе среза (`buffer = buffer[:0]`) гарантирует отсутствие аллокаций памяти в куче во время непрерывной работы батчера. Все события циркулируют через канал с емкостью буфера, что изолирует HTTP-хэндлеры от пауз сборщика мусора и сетевых задержек ClickHouse.

## Подводные камни

Если ClickHouse временно недоступен, канал батчера может заполниться. Если метод `Push` блокирующий (`b.ch <- e`), это парализует все входящие HTTP-запросы сервиса. Необходимо реализовывать либо вытеснение (Drop oldest), либо fallback на запись в локальный диск / Kafka.

## На собеседовании

**Вопрос:** Что произойдет с данными в памяти буферизатора при аварийном падении процесса (SIGKILL, OOM)? Как обеспечить Zero Data Loss?
**Ответ:** Данные в памяти Go-процесса будут утеряны. Для гарантии надежности в BigTech события сначала записываются во внешнюю персистентную шину (Kafka с репликацией и подтверждением acks=all), а Go-консьюмер вычитывает сообщения из Kafka, батчит их и сдвигает commit offset только после успешного `batch.Send()` в ClickHouse.
