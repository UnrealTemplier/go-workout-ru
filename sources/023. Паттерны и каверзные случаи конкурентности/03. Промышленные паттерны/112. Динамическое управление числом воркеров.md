# Динамическое управление числом воркеров через командный канал (Command Channel)

## Условие

Динамическое управление числом воркеров: main может добавлять или убирать воркеров через каналы команд.

## Теория

Паттерн Command Control: диспетчер слушает команды `ADD_WORKER` и `REMOVE_WORKER`, динамически запуская и останавливая горутины-воркеры.

## Решение по шагам

1. Создадим перечисление команд `CmdAdd` и `CmdRemove`.
2. Реализуем диспетчер с хранилищем каналов отмены `map[int]chan struct{}`.
3. Отправим команды добавления и удаления.
4. Проверим адаптацию пула.

## Код решения

**`main.go`**

```go
package main

import (
	"fmt"
	"time"
)

type Command int

const (
	CmdAdd Command = iota
	CmdRemove
)

type WorkerManager struct {
	cmdChan chan Command
	workers map[int]chan struct{}
	nextID  int
}

func NewWorkerManager() *WorkerManager {
	return &WorkerManager{
		cmdChan: make(chan Command),
		workers: make(map[int]chan struct{}),
		nextID:  1,
	}
}

func (m *WorkerManager) Start() {
	go func() {
		for cmd := range m.cmdChan {
			switch cmd {
			case CmdAdd:
				id := m.nextID
				m.nextID++
				stop := make(chan struct{})
				m.workers[id] = stop

				go func(workerID int, stopChan <-chan struct{}) {
					fmt.Printf(" [+] Воркер #%d успешно запущен\n", workerID)
					for {
						select {
						case <-stopChan:
							fmt.Printf(" [-] Воркер #%d остановлен по команде\n", workerID)
							return
						default:
							time.Sleep(10 * time.Millisecond)
						}
					}
				}(id, stop)

			case CmdRemove:
				if len(m.workers) > 0 {
					for id, stop := range m.workers {
						close(stop)
						delete(m.workers, id)
						break
					}
				}
			}
		}
	}()
}

func main() {
	mgr := NewWorkerManager()
	mgr.Start()

	// Добавляем 2 воркеров
	mgr.cmdChan <- CmdAdd
	mgr.cmdChan <- CmdAdd
	time.Sleep(20 * time.Millisecond)

	// Удаляем 1 воркера
	mgr.cmdChan <- CmdRemove
	time.Sleep(20 * time.Millisecond)

	close(mgr.cmdChan)
	fmt.Println("[Main] Управление пулом завершено.")
}
```

**`Терминал`**

```bash
go run main.go
# Вывод:
#  [+] Воркер #1 успешно запущен
#  [+] Воркер #2 успешно запущен
#  [-] Воркер #1 остановлен по команде
# [Main] Управление пулом завершено.
```

## Под капотом

Закрытие индивидуального канала `stop` останавливает конкретного воркера.

## Подводные камни

Всегда освобождайте закрытые каналы из мапы `delete(m.workers, id)`.

## На собеседовании

**Вопрос с собеседования:** «В чем преимущество передачи команд через канал перед вызовом методов структуры?»
**Ответ:** Канал команд сериализует все операции в единой горутине-диспетчере (Actor Model Pattern), устраняя необходимость в мьютексах для защиты внутреннего состояния пула.
