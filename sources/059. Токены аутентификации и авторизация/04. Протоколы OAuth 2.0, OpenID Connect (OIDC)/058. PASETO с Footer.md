# PASETO с Footer: Аутентифицированные метаданные и Key ID (kid)

## Условие

Добавьте в токен footer (незашифрованные, но аутентифицированные данные), например, `kid` (Key ID). При проверке токена убедитесь, что footer совпадает с ожидаемым.

### Исходная формулировка

**PASETO с footer.**: Добавьте в токен footer (незашифрованные, но аутентифицированные данные), например, `kid` (Key ID). При проверке токена убедитесь, что footer совпадает с ожидаемым.

## Теория

Формат **PASETO** поддерживает специальную секцию — **Footer (Нижний колонтитул)**:
$$\text{v4.public.} + \text{Base64URL}(\text{Payload} + \text{Signature}) + \text{"."} + \text{Base64URL}(\text{Footer})$$

### Криптографические свойства Footer:
1. **Открытость (Unencrypted):** Footer передается в открытом виде (Base64URL) и может быть прочитан любой стороной до верификации полезной нагрузки. Это идеальное место для передачи метаданных маршрутизации: `kid` (Key ID), идентификатора арендатора (`tenant_id`), или отпечатка сертификата.
2. **Аутентифицированность (Authenticated):** Footer **включен в Pre-Authentication Encoding (PAE)**:
   $$\text{M} = \text{PAE}(\text{Header}, \text{Payload}, \text{Footer})$$
   Цифровая подпись Ed25519 вычисляется над массивом \(M\), включающим Footer!

**Почему это исключает уязвимости?**
Если злоумышленник попытается подменить `kid` в Footer (например, чтобы сервер попытался проверить токен другим ключом), проверка подписи `ed25519.Verify` мгновенно вернет `false`! Footer невозможно изменить или подделать без знания приватного ключа.

## Решение по шагам

1. Создайте структуру Footer с полем `KeyID`.
2. Реализуйте функцию выпуска токена `SignV4PublicWithFooter` с включением footer в PAE.
3. Реализуйте верификатор, который декодирует Footer, находит соответствующий открытый ключ в пуле ключей (`KeyRing`) и проверяет подпись токена.
4. Продемонстрируйте успешную валидацию токена с динамическим выбором ключа по `kid`.
5. Продемонстрируйте сбой проверки при попытке модификации `kid` атакующим.

## Код решения

**`main.go`**

```go
package main

import (
	"crypto/ed25519"
	"crypto/rand"
	"encoding/base64"
	"encoding/binary"
	"encoding/json"
	"errors"
	"fmt"
	"strings"
	"time"
)

// PAE (Pre-Authentication Encoding) с поддержкой трех компонентов
func pae(pieces ...[]byte) []byte {
	buf := make([]byte, 8)
	binary.LittleEndian.PutUint64(buf, uint64(len(pieces)))
	out := append([]byte{}, buf...)
	for _, piece := range pieces {
		lenBuf := make([]byte, 8)
		binary.LittleEndian.PutUint64(lenBuf, uint64(len(piece)))
		out = append(out, lenBuf...)
		out = append(out, piece...)
	}
	return out
}

type PasetoFooter struct {
	KeyID string `json:"kid"`
}

type TokenPayload struct {
	Subject string `json:"sub"`
	Role    string `json:"role"`
	Exp     int64  `json:"exp"`
}

// SignV4PublicWithFooter создает токен с аутентифицированным footer
func SignV4PublicWithFooter(payload TokenPayload, footer PasetoFooter, privKey ed25519.PrivateKey) (string, error) {
	header := "v4.public."
	payloadBytes, _ := json.Marshal(payload)
	footerBytes, _ := json.Marshal(footer)

	// PAE(header, payload, footer)
	m := pae([]byte(header), payloadBytes, footerBytes)
	sig := ed25519.Sign(privKey, m)

	tokenBody := append(payloadBytes, sig...)
	token := fmt.Sprintf("%s%s.%s",
		header,
		base64.RawURLEncoding.EncodeToString(tokenBody),
		base64.RawURLEncoding.EncodeToString(footerBytes),
	)
	return token, nil
}

// VerifyV4PublicWithFooter извлекает kid, выбирает ключ и валидирует подпись
func VerifyV4PublicWithFooter(token string, keyRing map[string]ed25519.PublicKey) (*TokenPayload, *PasetoFooter, error) {
	const header = "v4.public."
	if !strings.HasPrefix(token, header) {
		return nil, nil, errors.New("неверный формат токена PASETO")
	}

	trimmed := strings.TrimPrefix(token, header)
	parts := strings.Split(trimmed, ".")
	if len(parts) != 2 {
		return nil, nil, errors.New("ожидалось 2 сегмента: payload_sig и footer")
	}

	rawBody, err := base64.RawURLEncoding.DecodeString(parts[0])
	if err != nil || len(rawBody) < ed25519.SignatureSize {
		return nil, nil, errors.New("невалидное тело токена")
	}

	footerBytes, err := base64.RawURLEncoding.DecodeString(parts[1])
	if err != nil {
		return nil, nil, errors.New("невалидный footer base64")
	}

	var footer PasetoFooter
	if err := json.Unmarshal(footerBytes, &footer); err != nil {
		return nil, nil, errors.New("ошибка парсинга footer")
	}

	// Выбор открытого ключа по kid из footer
	pubKey, exists := keyRing[footer.KeyID]
	if !exists {
		return nil, nil, fmt.Errorf("неизвестный идентификатор ключа (kid): %s", footer.KeyID)
	}

	payloadLen := len(rawBody) - ed25519.SignatureSize
	payload := rawBody[:payloadLen]
	sig := rawBody[payloadLen:]

	// Проверка подписи Ed25519 над PAE(header, payload, footer)
	m := pae([]byte(header), payload, footerBytes)
	if !ed25519.Verify(pubKey, m, sig) {
		return nil, nil, errors.New("подпись токена невалидна (возможно, подделан payload или footer!)")
	}

	var claims TokenPayload
	_ = json.Unmarshal(payload, &claims)

	return &claims, &footer, nil
}

func main() {
	pub1, priv1, _ := ed25519.GenerateKey(rand.Reader)
	pub2, _, _ := ed25519.GenerateKey(rand.Reader)

	keyRing := map[string]ed25519.PublicKey{
		"ed25519-key-2026-v1": pub1,
		"ed25519-key-2026-v2": pub2,
	}

	claims := TokenPayload{
		Subject: "usr_senior_dev_99",
		Role:    "cluster_admin",
		Exp:     time.Now().Add(1 * time.Hour).Unix(),
	}
	footer := PasetoFooter{KeyID: "ed25519-key-2026-v1"}

	token, err := SignV4PublicWithFooter(claims, footer, priv1)
	if err != nil {
		panic(err)
	}
	fmt.Printf("=== 1. Выпущенный PASETO v4 токен с Footer ===\n%s\n\n", token)

	// 2. Успешная валидация
	verifiedClaims, verifiedFooter, err := VerifyV4PublicWithFooter(token, keyRing)
	if err != nil {
		panic(err)
	}
	fmt.Println("=== 2. Успешная верификация с извлечением Footer ===")
	fmt.Printf("Key ID из Footer: %s\n", verifiedFooter.KeyID)
	fmt.Printf("Subject:          %s\n", verifiedClaims.Subject)
	fmt.Printf("Role:             %s\n\n", verifiedClaims.Role)

	// 3. Атака: подмена kid в Footer на v2
	tamperedFooterJSON := `{"kid":"ed25519-key-2026-v2"}`
	tamperedFooterB64 := base64.RawURLEncoding.EncodeToString([]byte(tamperedFooterJSON))
	parts := strings.Split(token, ".")
	tamperedToken := parts[0] + "." + parts[1] + "." + parts[2] + "." + tamperedFooterB64

	_, _, errTamper := VerifyV4PublicWithFooter(tamperedToken, keyRing)
	fmt.Println("=== 3. Попытка подмены kid в Footer злоумышленником ===")
	fmt.Printf("Результат: %v\n", errTamper)
}
```

## Под капотом

Спецификация PASETO гарантирует криптографическую целостность Footer за счет того, что байты Footer передаются третьим аргументом функции `pae(header, payload, footer)`. Таким образом, значение Footer хешируется непосредственно внутри алгоритма цифровой подписи Ed25519. Изменение хотя бы одного байта в JSON-структуре футера делает подпись недействительной, нейтрализуя атаки подмены ключа подписи.

## Подводные камни

1. **Помещение секретов в Footer:** Footer не шифруется ни в `v4.public`, ни в `v4.local`! Помещать туда пароли, API-ключи или PII-данные запрещено.
2. **Игнорирование ошибок декодирования Footer:** Если Footer поврежден, парсер обязан вернуть ошибку и прекратить обработку токена.
3. **Парсинг Payload до проверки Footer:** Полезная нагрузка должна оставаться непроверенными байтами до тех пор, пока не подтверждена подпись над связкой payload + footer.

## На собеседовании

**Вопрос с собеседования:** В каких случаях в Footer PASETO помещают не просто Key ID (kid), а публичный ключ отправителя или отпечаток сертификата?
**Ответ:** При реализации архитектур со **взаимной аутентификацией на прикладном уровне (Application-Layer Mutual Auth / DPoP style)**:
1) Клиент помещает в Footer отпечаток своего открытого ключа или открытый ключ эфемерной сессии.
2) Промежуточные шлюзы (API Gateways) могут маршрутизировать запрос на нужный сегмент сети или проверять права клиента до десериализации основного тела.
3) Поскольку Footer аутентифицирован подписью, промежуточный узел уверен в подлинности метаданных маршрутизации без риска инъекций.
