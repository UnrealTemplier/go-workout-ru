# Сессии и транзакции с откатом при отрицательном балансе

## Условие

Создай сессию (`client.StartSession`). В рамках транзакции (`session.WithTransaction`) переведи деньги между двумя документами в коллекции `accounts`. Если баланс отправителя уходит в минус — вызови `session.AbortTransaction()`.

### Исходная формулировка

**[Сессии и Транзакции]**: Создай сессию (`client.StartSession`). В рамках транзакции (`session.WithTransaction`) переведи деньги между двумя документами в коллекции `accounts`. Если баланс отправителя уходит в минус — вызови `session.AbortTransaction()`.

## Теория

Многодокументные транзакции с функцией обратного вызова `session.WithTransaction` являются идиоматичным стандартом Go.

### Гарантии `session.WithTransaction`:
1. Автоматически инициирует транзакцию `StartTransaction`.
2. Передает `mongo.SessionContext` в функцию обратного вызова.
3. При возврате ошибки из замыкания автоматически выполняет `AbortTransaction`.
4. При успешном возврате `nil` вызывает `CommitTransaction`.
5. При возникновении временных сетевых сбоев или конфликтов блокировок (`TransientTransactionError`, `UnknownTransactionCommitResult`) автоматически перезапускает попытку до истечения таймаута (120 сек).

## Решение по шагам

1. Создайте сессию через `client.StartSession()`.
2. Реализуйте коллбэк функции перевода с проверкой баланса.
3. Если баланс меньше суммы перевода — верните `fmt.Errorf`, что спровоцирует откат.
4. Протестируйте успешный и неуспешный сценарий перевода денег.

## Код решения

**`main.go`**

```go
package main

import (
	"context"
	"fmt"
	"time"

	"go.mongodb.org/mongo-driver/bson"
	"go.mongodb.org/mongo-driver/mongo"
	"go.mongodb.org/mongo-driver/mongo/options"
	"go.mongodb.org/mongo-driver/mongo/readconcern"
	"go.mongodb.org/mongo-driver/mongo/writeconcern"
)

func SafeTransfer(ctx context.Context, client *mongo.Client, from, to string, amount float64) error {
	session, err := client.StartSession()
	if err != nil {
		return err
	}
	defer session.EndSession(ctx)

	txnOpts := options.Transaction().
		SetReadConcern(readconcern.Snapshot()).
		SetWriteConcern(writeconcern.Majority())

	_, err = session.WithTransaction(ctx, func(sc mongo.SessionContext) (interface{}, error) {
		coll := client.Database("bank").Collection("accounts")

		// 1. Атомарно списываем только при условии достаточного баланса
		filterSender := bson.M{
			"acc_id":  from,
			"balance": bson.M{"$gte": amount}, // Проверка на уровне условия БД
		}
		res, err := coll.UpdateOne(sc, filterSender, bson.M{"$inc": bson.M{"balance": -amount}})
		if err != nil {
			return nil, err
		}
		if res.MatchedCount == 0 {
			// Вызов возврата ошибки спровоцирует автоматический AbortTransaction
			return nil, fmt.Errorf("транзакция отклонена: на счете %s недостаточно средств для списания %.2f", from, amount)
		}

		// 2. Начисляем получателю
		_, err = coll.UpdateOne(sc, bson.M{"acc_id": to}, bson.M{"$inc": bson.M{"balance": amount}})
		if err != nil {
			return nil, err
		}

		return nil, nil
	}, txnOpts)

	return err
}

func main() {
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()

	client, err := mongo.Connect(ctx, options.Client().ApplyURI("mongodb://localhost:27017/?replicaSet=rs0"))
	if err != nil {
		panic(err)
	}
	defer client.Disconnect(ctx)

	coll := client.Database("bank").Collection("accounts")
	_ = coll.Drop(ctx)

	_, _ = coll.InsertMany(ctx, []interface{}{
		bson.M{"acc_id": "ACC-USER-1", "balance": 150.0},
		bson.M{"acc_id": "ACC-USER-2", "balance": 50.0},
	})

	// Сценарий 1: перевод превышает баланс
	fmt.Println("Попытка перевести 200 руб со счета с балансом 150 руб:")
	err = SafeTransfer(ctx, client, "ACC-USER-1", "ACC-USER-2", 200.0)
	if err != nil {
		fmt.Printf("Откат: %v\n", err)
	}

	var acc1 bson.M
	_ = coll.FindOne(ctx, bson.M{"acc_id": "ACC-USER-1"}).Decode(&acc1)
	fmt.Printf("Баланс отправителя остался неизменным: %.2f руб.\n\n", acc1["balance"])

	// Сценарий 2: легитимный перевод
	fmt.Println("Попытка перевести 50 руб:")
	err = SafeTransfer(ctx, client, "ACC-USER-1", "ACC-USER-2", 50.0)
	if err != nil {
		panic(err)
	}
	_ = coll.FindOne(ctx, bson.M{"acc_id": "ACC-USER-1"}).Decode(&acc1)
	fmt.Printf("Успешный коммит! Новый баланс отправителя: %.2f руб.\n", acc1["balance"])
}
```

## Под капотом

Snapshot изоляция в WiredTiger использует мультиверсионное управление параллельным доступом (MVCC). Когда транзакция стартует, она получает метку времени snapshot. Любые изменения, закоммиченные другими сессиями после этой метки, не видны внутри текущей транзакции до ее завершения.

## Подводные камни

Если внутри замыкания `WithTransaction` вызывать ручные методы `session.AbortTransaction(sc)` или `session.CommitTransaction(sc)`, произойдет конфликт состояний машины состояний сессии. Замыкание должно лишь возвращать ошибку или `nil`, позволяя `WithTransaction` управлять жизненным циклом коммита.

## На собеседовании

**Вопрос:** Что делает драйвер MongoDB при получении ошибки TransientTransactionError?
**Ответ:** Драйвер понимает, что транзакция прервалась из-за временного сетевого сбоя или конфликта параллельной записи (Write Conflict). Функция `session.WithTransaction` не возвращает ошибку наверх, а автоматически выполняет повторный вызов коллбэка с чистого листа, пытаясь успешно закоммитить транзакцию в течение retry-таймаута.
