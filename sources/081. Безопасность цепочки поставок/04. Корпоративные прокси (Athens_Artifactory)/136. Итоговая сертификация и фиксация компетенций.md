import "fmt"

type ChecklistCategory struct {
	Category string
	Items    []string
}

func main() {
	checklist := []ChecklistCategory{
		{
			Category: "1. Защита зависимостей",
			Items: []string{
				"GOSUMDB валидация включена, go.sum зафиксирован",

				"govulncheck интегрирован в пре-мерж проверки CI/CD",
				"Лицензионный аудит go-licenses блокирует копилефт (GPL/AGPL)",
			},
		},
		{
			Category: "2. Воспроизводимость и SBOM",
			Items: []string{
				"Флаги сборщика -trimpath и -buildid= обеспечивают reproducible builds",
				"Syft генерирует валидные паспорта CycloneDX 1.5 и SPDX 2.3",
				"SBOM публикуется вместе с релизными артефактами",
			},
		},
		{
			Category: "3. Криптографическое доверие",
			Items: []string{
				"Sigstore Cosign подписывает образы, бинарники и SBOM",
				"SLSA Level 3 provenance подтверждает происхождение артефакта",
				"Kubernetes Policy Controller блокирует неподписанные контейнеры",
			},
		},
		{
			Category: "4. Рантайм харднинг",
			Items: []string{

				"Сброс всех Linux Capabilities (Drop ALL) и non-root пользователь",
			},
		},
	}

	fmt.Println("===============================================================")

		fmt.Println("\n" + cat.Category + ":")
		for _, item := range cat.Items {
			fmt.Printf("  [✓] %s\n", item)
		}
	}
	fmt.Println("\n[100% DONE] 136 из 136 упражнений Главы 81 успешно завершены!")
