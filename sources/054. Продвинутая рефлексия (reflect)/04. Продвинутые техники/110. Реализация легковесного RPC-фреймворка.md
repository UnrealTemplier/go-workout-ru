# Реализация легковесного RPC-фреймворка через reflect

## Условие

Создайте простой RPC фреймворк, который автоматически регистрирует методы структуры как RPC endpoints.

### Исходная формулировка

**RPC framework через reflect**: Создайте простой RPC фреймворк, который автоматически регистрирует методы структуры как RPC endpoints.

## Теория

Стандартный пакет `net/rpc` использует строгую конвенцию рефлексии:
1. Метод обязан быть экспортированным (начинаться с заглавной буквы).
2. Метод должен иметь ровно два аргумента: входной (`args`) и выходной указатель (`reply`).
3. Метод должен возвращать ровно одно значение типа `error`.
Сигнатура: `func (t *Service) MethodName(args ReqType, reply *RespType) error`.

RPC сервер:
- Инспектирует структуру через `reflect.TypeOf(svc)` и находит подходящие методы.
- Сохраняет дескриптор метода: типы аргументов и ответов.
- При получении текстового/бинарного запроса десериализует данные в `reflect.New(reqType)`, вызывает метод через `method.Func.Call(...)`, и сериализует `reply` обратно клиенту.

## Решение по шагам

1. Объявите конвенцию сигнатуры RPC метода.
2. Создайте `RPCServer` со словарем `services map[string]map[string]*MethodDesc`.
3. Напишите метод `Register(rcvr any) error`.
4. Реализуйте метод `Call(serviceMethod string, rawJSON string) (string, error)`.
5. Протестируйте вызов сервиса `MathService.Multiply`.

## Код решения

**`main.go`**

```go
package main

import (
	"encoding/json"
	"fmt"
	"reflect"
	"strings"
)

type MethodDesc struct {
	Method  reflect.Method
	ArgType reflect.Type
	RepType reflect.Type
}

type RPCServer struct {
	services map[string]map[string]*MethodDesc
	receivers map[string]reflect.Value
}

func NewRPCServer() *RPCServer {
	return &RPCServer{
		services:  make(map[string]map[string]*MethodDesc),
		receivers: make(map[string]reflect.Value),
	}
}

func (s *RPCServer) Register(rcvr any) error {
	rcvrVal := reflect.ValueOf(rcvr)
	rcvrType := rcvrVal.Type()
	serviceName := reflect.Indirect(rcvrVal).Type().Name()

	methods := make(map[string]*MethodDesc)

	for i := 0; i < rcvrType.NumMethod(); i++ {
		m := rcvrType.Method(i)
		mType := m.Type

		// Проверяем конвенцию RPC: 3 аргумента (receiver, args, reply), 1 возврат (error)
		if mType.NumIn() != 3 || mType.NumOut() != 1 {
			continue
		}

		if mType.Out(0) != reflect.TypeOf((*error)(nil)).Elem() {
			continue
		}

		argType := mType.In(1)
		repType := mType.In(2)

		// Reply обязан быть указателем
		if repType.Kind() != reflect.Pointer {
			continue
		}

		methods[m.Name] = &MethodDesc{
			Method:  m,
			ArgType: argType,
			RepType: repType.Elem(),
		}
	}

	if len(methods) == 0 {
		return fmt.Errorf("сервис %s не содержит валидных RPC методов", serviceName)
	}

	s.services[serviceName] = methods
	s.receivers[serviceName] = rcvrVal
	return nil
}

func (s *RPCServer) Execute(serviceDotMethod string, jsonArgs string) (string, error) {
	parts := strings.Split(serviceDotMethod, ".")
	if len(parts) != 2 {
		return "", fmt.Errorf("неверный формат RPC вызова, ожидался 'Service.Method'")
	}

	svcName, methodName := parts[0], parts[1]
	svc, ok := s.services[svcName]
	if !ok {
		return "", fmt.Errorf("сервис %s не найден", svcName)
	}

	mDesc, ok := svc[methodName]
	if !ok {
		return "", fmt.Errorf("метод %s не найден в сервисе %s", methodName, svcName)
	}

	// Десериализуем входные аргументы
	argVal := reflect.New(mDesc.ArgType)
	if err := json.Unmarshal([]byte(jsonArgs), argVal.Interface()); err != nil {
		return "", fmt.Errorf("ошибка парсинга аргументов: %w", err)
	}

	// Создаем буфер для ответа
	replyVal := reflect.New(mDesc.RepType)

	// Вызов метода: receiver, argVal.Elem(), replyVal
	rcvrVal := s.receivers[svcName]
	results := mDesc.Method.Func.Call([]reflect.Value{rcvrVal, argVal.Elem(), replyVal})

	errInter := results[0].Interface()
	if errInter != nil {
		return "", errInter.(error)
	}

	respBytes, err := json.Marshal(replyVal.Interface())
	if err != nil {
		return "", err
	}
	return string(respBytes), nil
}

// Демонстрационный RPC-сервис
type OrderService struct{}

type OrderRequest struct {
	OrderID  string `json:"order_id"`
	Quantity int    `json:"quantity"`
}

type OrderResponse struct {
	Status string  `json:"status"`
	Total  float64 `json:"total"`
}

func (OrderService) ProcessOrder(req OrderRequest, resp *OrderResponse) error {
	if req.Quantity <= 0 {
		return fmt.Errorf("количество товара должно быть > 0")
	}
	resp.Status = "CONFIRMED"
	resp.Total = float64(req.Quantity) * 1250.50
	return nil
}

func main() {
	server := NewRPCServer()
	server.Register(&OrderService{})

	// Эмулируем входящий сетевой RPC-запрос в формате JSON
	reqJSON := `{"order_id":"ORD-9901","quantity":4}`
	respJSON, err := server.Execute("OrderService.ProcessOrder", reqJSON)
	if err != nil {
		panic(err)
	}

	fmt.Println("RPC Ответ сервера:")
	fmt.Println(respJSON)
}
```

ℹ️ Архитектура RPC-сервера со сканированием методов и динамической диспетчеризацией

**`Терминал`**

```bash
go run main.go
# Вывод:
# RPC Ответ сервера:

```

## Под капотом

Спецификация `net/rpc` использует `mType.Out(0) != reflect.TypeOf((*error)(nil)).Elem()`, чтобы проверить, реализует ли тип интерфейс ошибки `error`. При вызове `mDesc.Method.Func.Call` первым параметром передается получатель `rcvrVal`. Выделение памяти под ответ через `reflect.New(mDesc.RepType)` гарантирует, что метод получает инициализированный указатель на чистую структуру ответа.

## Подводные камни

1. Забытое разыменование указателя на `error`: `reflect.TypeOf(error(nil))` вернет `nil`, необходимо использовать `reflect.TypeOf((*error)(nil)).Elem()`.
2. Аргумент `reply` не является указателем: сервис не сможет записать данные ответа.
3. Паника внутри RPC-метода: без `recover` в цикле диспетчеризации упадет весь RPC сервер.

## На собеседовании

**Вопрос с собеседования:** Почему в современной BigTech микросервисной архитектуре Go RPC (`net/rpc`) был вытеснен gRPC?
**Ответ:** `net/rpc` использует собственный текстовый/гоб протокол поверх raw TCP, требует рефлексии в рантайме и специфичен для Go (полиглотность отсутствует). gRPC основан на HTTP/2 (мультиплексирование, стриминг, заголовки, flow control), использует компактный бинарный Protocol Buffers, кросс-язычен и полностью генерирует код сериализации и маршрутизации на этапе компиляции, исключая рефлексивные задержки.
